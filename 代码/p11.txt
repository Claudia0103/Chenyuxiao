import os
import numpy as np
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
from matplotlib import font_manager as fm
from matplotlib.patches import FancyBboxPatch
try:
    HERE = os.path.dirname(os.path.abspath(__file__))
except NameError:
    HERE = os.path.abspath(os.getcwd())
OUT = os.path.join(HERE, "png")
os.makedirs(OUT, exist_ok=True)
DPI = 120.0
EM_PT = DPI / 72.0
SIZE_PX = [640, 520]
BG = "#1A1E43"
ORANGE = "#ED7D31"
DASH = "#ED7D31"
BLUE_TAG = "#9DD9F3"
ORANGE_TAG = "#F5B041"
TXT = "#FFFFFF"
LBL = "#F2F2F2"
AXIS_TXT = "#ED7D31"
NOTE = "#BFBFBF"
AXIS_LINE = "#8FAADC"
MONTHS = ["5月", "6月", "7月", "8月", "9月", "10月", "11月", "12月",
          "1月", "2月", "3月"]
SALES = [146, 198, 296, 412, 506, 615, 789, 1021, 3782, 3215, 2936]
_FONT_FILES = {
    "ea": {"n": "C:/Windows/Fonts/msyh.ttc", "b": "C:/Windows/Fonts/msyhbd.ttc"},
    "cal": {"n": "C:/Windows/Fonts/calibri.ttf", "b": "C:/Windows/Fonts/calibrib.ttf"},
    "num": {"n": "C:/Windows/Fonts/arial.ttf", "b": "C:/Windows/Fonts/arialbd.ttf"},
}
for _d in _FONT_FILES.values():
    for _p in _d.values():
        try:
            fm.fontManager.addfont(_p)
        except Exception:
            pass
_FP_CACHE = {}
def fp(sz, bold=False, fam="ea"):
    key = (round(float(sz), 2), bool(bold), fam)
    p = _FP_CACHE.get(key)
    if p is None:
        p = fm.FontProperties(fname=_FONT_FILES[fam]["b" if bold else "n"])
        p.set_size(sz)
        _FP_CACHE[key] = p
    return p
def smooth_curve(x, y, n=400):
    x2 = np.append(x, x[-1] + (x[-1] - x[-2]))
    y2 = np.append(y, y[-1] + 1.15 * (y[-1] - y[-2]))
    xs = np.linspace(x.min(), x2.max(), n)
    try:
        from scipy.interpolate import make_interp_spline
        spl = make_interp_spline(x2, y2, k=3)
        ys = spl(xs)
    except Exception:
        ys = np.interp(xs, x, y)
    mask = xs <= x.max()
    return xs[mask], ys[mask]
def draw():
    W, H = SIZE_PX
    fig = plt.figure(figsize=(W / DPI, H / DPI), dpi=DPI)
    fig.patch.set_facecolor(BG)
    ax = fig.add_axes([0, 0, 1, 1])
    ax.set_xlim(0, W)
    ax.set_ylim(0, H)
    ax.set_facecolor(BG)
    for side in ("top", "right", "bottom", "left"):
        ax.spines[side].set_visible(False)
    ax.set_xticks([])
    ax.set_yticks([])
    PLOT_L = 78.0
    PLOT_R = 610.0
    PLOT_B = 175.0
    PLOT_T = 340.0
    VMAX = 4200.0
    n = len(MONTHS)
    step = (PLOT_R - PLOT_L) / float(n)
    def x_center(i):
        return PLOT_L + step * (i + 0.5)
    def x_edge(i):
        return PLOT_L + step * i
    def y_at(v):
        return PLOT_B + v / VMAX * (PLOT_T - PLOT_B)
    x = np.array([x_center(i) for i in range(n)], dtype=float)
    y = np.array([y_at(v) for v in SALES], dtype=float)
    xs, ys = smooth_curve(x, y, n=500)
    ax.plot(xs, ys, color=ORANGE, linewidth=2.0, zorder=4)
    idx_top = int(np.argmax(ys))
    x_top = xs[idx_top]
    y_top = ys[idx_top]

    ax.plot(
        [x_top, x_top], [PLOT_B, y_top],
        color=DASH, linewidth=1.2, linestyle=(0, (4, 4)), zorder=3,
    )
    ax.text(
        x_top, y_top + 6.0, "%d" % SALES[int(np.argmax(SALES))],
        fontproperties=fp(10.0, False, "num"),
        color=TXT, ha="center", va="bottom", zorder=7,
    )
    ax.plot([PLOT_L, PLOT_R], [PLOT_B, PLOT_B], color=AXIS_LINE, linewidth=0.8, zorder=2)
    for i in range(n + 1):
        xe = x_edge(i)
        ax.plot(
            [xe, xe], [PLOT_B - 5.0, PLOT_B],
            color=AXIS_LINE, linewidth=0.9, zorder=3,
        )
    for i in range(n):
        ax.text(
            x_center(i), PLOT_B - 16.0, MONTHS[i],
            fontproperties=fp(9.5, False, "ea"),
            color=LBL, ha="center", va="top", zorder=7,
        )
    for v in [0, 1000, 2000, 3000, 4000]:
        yy = y_at(v)
        ax.text(
            PLOT_L - 14.0, yy, "{:,}".format(v),
            fontproperties=fp(9.0, False, "num"),
            color=AXIS_TXT, ha="right", va="center", zorder=7,
        )
    tag_y = PLOT_B - 72.0
    tag_h = 24.0
    tag_pad = 4.0
    tag_r = 10.0
    x_2021_0 = x_edge(0) - 2.0
    x_2021_1 = x_edge(8) - 12.0
    w1 = x_2021_1 - x_2021_0
    ax.add_patch(FancyBboxPatch(
        (x_2021_0 + tag_pad, tag_y + tag_pad),
        w1 - 2 * tag_pad, tag_h - 2 * tag_pad,
        boxstyle="round,pad=%f,rounding_size=%f" % (tag_pad, tag_r),
        facecolor=BLUE_TAG, edgecolor="none", zorder=4,
        mutation_aspect=1.0,
    ))
    ax.text(
        (x_2021_0 + x_2021_1) / 2.0, tag_y + tag_h / 2.0, "2021",
        fontproperties=fp(11.0, False, "ea"),
        color="#1A1E43", ha="center", va="center", zorder=8,
    )
    x_2022_0 = x_edge(8) + 12.0
    x_2022_1 = x_edge(n) - 2.0
    w2 = x_2022_1 - x_2022_0
    ax.add_patch(FancyBboxPatch(
        (x_2022_0 + tag_pad, tag_y + tag_pad),
        w2 - 2 * tag_pad, tag_h - 2 * tag_pad,
        boxstyle="round,pad=%f,rounding_size=%f" % (tag_pad, tag_r),
        facecolor=ORANGE_TAG, edgecolor="none", zorder=4,
        mutation_aspect=1.0,
    ))
    ax.text(
        (x_2022_0 + x_2022_1) / 2.0, tag_y + tag_h / 2.0, "2022",
        fontproperties=fp(11.0, False, "ea"),
        color="#1A1E43", ha="center", va="center", zorder=8,
    )
    fig.text(
        0.048, 1.0 - (26.0) / H, "化妆品品类月度销量走势",
        fontproperties=fp(18.0, True, "ea"),
        color=TXT, ha="left", va="top", zorder=30,
    )
    fig.text(
        0.048, 1.0 - (62.0) / H, "2022年销量迅速增加，1月最高，销量达到3782",
        fontproperties=fp(11.0, False, "ea"),
        color=TXT, ha="left", va="top", zorder=30,
    )
    fig.text(
        0.048, 1.0 - (500.0) / H, "*注：数据来源于公司销售系统，统计日期截至2022.03.31",
        fontproperties=fp(8.5, False, "ea"),
        color=NOTE, ha="left", va="top", zorder=30,
    )
    return fig
if __name__ == "__main__":
    f = draw()
    out = os.path.join(OUT, "chart11_平滑折线图.png")
    f.savefig(out, dpi=DPI, facecolor=BG)
    try:
        from IPython import get_ipython
        if get_ipython() is not None:
            from IPython.display import display
            display(f)
    except Exception:
        pass
    plt.close(f)
    print("已输出：", out)
