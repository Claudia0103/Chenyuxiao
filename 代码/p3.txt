_CJK_RE = re.compile(r"[\u3000-\u303F\u4E00-\u9FFF\uFF00-\uFF60]")
_FP_CACHE = {}
def fp(sz, bold=False, ea=False):
    key = (round(float(sz), 2), bool(bold), bool(ea))
    p = _FP_CACHE.get(key)
    if p is None:
        p = fm.FontProperties(fname=_FONT_FILES["ea" if ea else "lt"]["b" if bold else "n"])
        p.set_size(sz)
        _FP_CACHE[key] = p
    return p
def mixed(target, x, y, s, fontsize=8, color=LBL, bold=False,
          ha="center", va="center", zorder=10):
    spans = []
    for ch in str(s):
        cjk = bool(_CJK_RE.match(ch))
        if spans and spans[-1][1] == cjk:
            spans[-1][0] += ch
        else:
            spans.append([ch, cjk])
    if len(spans) == 1:
        t, cjk = spans[0]
        return [target.text(x, y, t, fontproperties=fp(fontsize, bold, cjk),
                            color=color, ha=ha, va=va, zorder=zorder)]
    renderer = target.figure.canvas.get_renderer()
    ws = []
    for t, cjk in spans:
        a = target.text(0, 0, t, fontproperties=fp(fontsize, bold, cjk), alpha=0)
        ws.append(a.get_window_extent(renderer).width)
        a.remove()
    tr = target.transData
    ax_, ay_ = tr.transform((x, y))
    x0 = ax_ - sum(ws) / 2.0 if ha == "center" else ax_
    inv = tr.inverted()
    out = []
    for (t, cjk), w in zip(spans, ws):
        fx, fy = inv.transform((x0 + w / 2.0, ay_))
        out.append(target.text(fx, fy, t, fontproperties=fp(fontsize, bold, cjk),
                               color=color, ha="center", va=va, zorder=zorder))
        x0 += w
    return out
def fmtk(x, _=None):
    return "%d" % int(round(x))
def bar_path(x, w, bottom, top, rx_t, ry_t, rx_b, ry_b):
    k = 0.5523
    x0, x1 = x - w / 2.0, x + w / 2.0
    verts = [
        (x0, bottom + ry_b),
        (x0, top - ry_t),
        (x0, top - ry_t + k * ry_t), (x0 + rx_t - k * rx_t, top), (x0 + rx_t, top),
        (x1 - rx_t, top),
        (x1 - rx_t + k * rx_t, top), (x1, top - ry_t + k * ry_t), (x1, top - ry_t),
        (x1, bottom + ry_b),
        (x1, bottom + ry_b - k * ry_b), (x1 - rx_b + k * rx_b, bottom), (x1 - rx_b, bottom),
        (x0 + rx_b, bottom),
        (x0 + rx_b - k * rx_b, bottom), (x0, bottom + ry_b - k * ry_b), (x0, bottom + ry_b),
        (x0, bottom + ry_b),
    ]
    codes = [Path.MOVETO, Path.LINETO,
             Path.CURVE4, Path.CURVE4, Path.CURVE4,
             Path.LINETO,
             Path.CURVE4, Path.CURVE4, Path.CURVE4,
             Path.LINETO,
             Path.CURVE4, Path.CURVE4, Path.CURVE4,
             Path.LINETO,
             Path.CURVE4, Path.CURVE4, Path.CURVE4,
             Path.CLOSEPOLY]
    return Path(verts, codes)
def grad_bar(ax, x, h, w, rx_px, ry_px, zorder=3, n=256):
    t = np.linspace(0.0, 1.0, n)
    tt = np.array([g[0] for g in GRAD])
    img = np.empty((n, 1, 3), dtype=np.float32)
    for ch, ci in enumerate((1, 2, 3)):
        img[:, 0, ch] = np.interp(t, tt, np.array([g[ci] for g in GRAD], dtype=np.float32))
    img /= 255.0
    im = ax.imshow(img, extent=[x - w / 2.0, x + w / 2.0, 0.0, h],
                   origin="upper", aspect="auto", zorder=zorder,
                   interpolation="bilinear")
    xa, xb = ax.get_xlim()
    ya, yb = ax.get_ylim()
    bb = ax.get_window_extent()
    rx = rx_px * (xb - xa) / bb.width
    ry = ry_px * (yb - ya) / bb.height
    clip = PathPatch(bar_path(x, w, 0.0, h, rx, ry, rx, ry),
                     transform=ax.transData, facecolor="none", edgecolor="none")
    im.set_clip_path(clip)
    return im

BOX_TITLE = [0.05116, 0.03399, 0.96507, 0.22315]

INSET_PT = (0.05, 0.05, 0.1, 0.1)
TB_TWEAK = {0: [-12.5, -0.6], 1: [-0.4]}
def draw_textboxes(fig, W, H):
    renderer = fig.canvas.get_renderer()
    SC = DPI / float(fig.dpi)
    def measure(text, sz, bold):
        a = fig.text(0, 0, text, fontproperties=fp(sz, bold, ea=True),
                     alpha=0, zorder=-1)
        w = a.get_window_extent(renderer).width * SC
        a.remove()
        return w
    boxes = [
        (BOX_TITLE, TITLE_RUNS),
        (BOX_NOTE, [NOTE_PARA]),
    ]
    for bi, (box, paras) in enumerate(boxes):
        x0, y0, x1, y1 = box
        lI = INSET_PT[2] * DPI
        tI = INSET_PT[0] * DPI
        xL = x0 * W + lI
        y = y0 * H + tI
        for pi, p in enumerate(paras):
            sz, bold = p["sz"], p["bold"]
            em = sz * EM_PT
            lh = (p.get("lnspc") or 1.71 * sz) * EM_PT
            tw = TB_TWEAK.get(bi, [])
            off = tw[pi] if pi < len(tw) else 0.0
            ytop = y + 0.43 * em + off
            x = xL
            rdx = p.get("run_dx") or []
            for ri, text in enumerate(p["runs"]):
                xr = x + (rdx[ri] if ri < len(rdx) else 0.0)
                fig.text(xr / W, 1.0 - ytop / H, text,
                         fontproperties=fp(sz, bold, ea=True),
                         color=p["color"], ha="left", va="top", zorder=30)
                x += measure(text, sz, bold)
            y += lh
DLAB = [
    (0.18076, 0.45536),
    (0.31587, 0.52416),
    (0.44638, 0.46765),
    (0.58150, 0.37920),
    (0.71444, 0.44195),

CARD_W, CARD_H = 47.0, 26.0
CARD_R = 4.2
CARD_DX, CARD_DY = 10.5, 14.5
DOT_MS = 5.4
LEAD_LW = 0.75
NUM_DX, NUM_DY = 29.5, 12.3
NUM_SZ = 7.2
XLAB_SZ = 9.0
XLAB_PAD = 7.8
YLAB_PAD = 8.4
def draw():
    W, H = SIZE_PX
    fig = plt.figure(figsize=(W / DPI, H / DPI), dpi=DPI)
    fig.patch.set_facecolor(BG)
    px, py, pw, ph = PLOT_PX
    ax = fig.add_axes([px, 1.0 - (py + ph), pw, ph])
    ax.set_facecolor(BG)
    ax.set_ylim(0, YMAX)
    ax.set_xlim(-0.5, len(VALS) - 0.5)
    for i, v in enumerate(VALS):
        grad_bar(ax, i, v, 1.0 / 6.0, rx_px=4.6, ry_px=4.6, zorder=3)
    ax.set_axisbelow(True)
    ax.grid(axis="y", color=GRID, linewidth=0.75,
            linestyle=(GRID_DASH_OFF, GRID_DASH))
    for side in ("top", "right", "left", "bottom"):
        ax.spines[side].set_visible(False)
    ax.set_yticks([0, 200, 400, 600, 800, 1000, 1200])
    ax.yaxis.set_major_formatter(FuncFormatter(fmtk))
    ax.tick_params(axis="y", length=0, pad=YLAB_PAD, colors=LBL)
    for lb in ax.get_yticklabels():
        lb.set_fontproperties(fp(NUM_SZ, False, False))
        lb.set_color(LBL)
    ax.set_xticks(range(len(CATS)))
    ax.tick_params(axis="x", length=0, pad=XLAB_PAD, colors=LBL)
    ax.set_xticklabels(CATS, fontsize=XLAB_SZ, color=LBL)
    for lb in ax.get_xticklabels():
        lb.set_fontproperties(fp(XLAB_SZ, False, True))
    inv = ax.transData.inverted()
    def to_data(xp, yp):
        return inv.transform((xp, H - yp))
    dxu = (ax.get_xlim()[1] - ax.get_xlim()[0]) / (PLOT_PX[2] * W)
    dyu = (ax.get_ylim()[1] - ax.get_ylim()[0]) / (PLOT_PX[3] * H)
    for i, v in enumerate(VALS):
        cx = DLAB[i][0] * W
        cy = DLAB[i][1] * H
        dotx_px, doty_px = cx + CARD_DX, cy + CARD_DY
        (bx0, by_top) = to_data(cx, cy)
        (bx1, by_bot) = to_data(cx + CARD_W, cy + CARD_H)
        rxp = CARD_R * dxu
        ryp = CARD_R * dyu
        pth = bar_path((bx0 + bx1) / 2.0, bx1 - bx0, by_bot, by_top,
                       rxp, ryp, rxp, ryp)
        ax.add_patch(PathPatch(pth, facecolor=BOX, edgecolor="none", zorder=5))
        (lx, ly) = to_data(dotx_px, doty_px)
        ax.plot([lx, lx], [v, ly], color=DOT, lw=LEAD_LW,
                solid_capstyle="butt", zorder=4)
        ax.plot([lx], [ly], "o", ms=DOT_MS, mfc=DOT, mec="none", zorder=6)
        (nx, ny) = to_data(cx + NUM_DX, cy + NUM_DY)
        ax.text(nx, ny, fmtk(v), fontproperties=fp(NUM_SZ, False, False),
                color=LBL, ha="center", va="center", zorder=7)
    draw_textboxes(fig, W, H)
    return fig
