import os
import numpy as np
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
from matplotlib import font_manager as fm
try:
    HERE = os.path.dirname(os.path.abspath(__file__))
except NameError:
    HERE = os.path.abspath(os.getcwd())
OUT = os.path.join(HERE, "png")
os.makedirs(OUT, exist_ok=True)
DPI = 120.0
EM_PT = DPI / 72.0
SIZE_PX = [640, 520]
BG = "#1A1E43"
TXT = "#FFFFFF"
NOTE = "#BFBFBF"
AXIS_LINE = "#8FAADC"
MONTHS = ["1月", "2月", "3月", "4月", "5月", "6月",
          "7月", "8月", "9月", "10月", "11月", "12月"]
VALUES = [2354, 1902, 3524, 2698, 2896, 2563,
          3156, 2895, 3621, 2635, 2963, 2789]
Q_TOTALS = [7780, 8157, 9673, 8387]
Q_BG_COLORS = ["#2A3A8C", "#7A3A10", "#6A5510", "#6A2050"]
Q_BAR_COLORS = ["#3E5BD6", "#F07010", "#F0B03F", "#E05590"]
_FONT_FILES = {
    "ea": {"n": "C:/Windows/Fonts/msyh.ttc", "b": "C:/Windows/Fonts/msyhbd.ttc"},
    "cal": {"n": "C:/Windows/Fonts/calibri.ttf", "b": "C:/Windows/Fonts/calibrib.ttf"},
    "num": {"n": "C:/Windows/Fonts/arial.ttf", "b": "C:/Windows/Fonts/arialbd.ttf"},
}
for _d in _FONT_FILES.values():
    for _p in _d.values():
        try:
            fm.fontManager.addfont(_p)
        except Exception:
            pass
_FP_CACHE = {}
def fp(sz, bold=False, fam="ea"):
    key = (round(float(sz), 2), bool(bold), fam)
    p = _FP_CACHE.get(key)
    if p is None:
        p = fm.FontProperties(fname=_FONT_FILES[fam]["b" if bold else "n"])
        p.set_size(sz)
        _FP_CACHE[key] = p
    return p
def draw():
    W, H = SIZE_PX
    fig = plt.figure(figsize=(W / DPI, H / DPI), dpi=DPI)
    fig.patch.set_facecolor(BG)
    ax = fig.add_axes([0, 0, 1, 1])
    ax.set_xlim(0, W)
    ax.set_ylim(0, H)
    ax.set_facecolor(BG)
    for side in ("top", "right", "bottom", "left"):
        ax.spines[side].set_visible(False)
    ax.set_xticks([])
    ax.set_yticks([])
    PLOT_L = 60.0
    PLOT_R = 610.0
    PLOT_B = 80.0
    PLOT_T = 350.0
    VMAX = 10500.0
    n = 12
    step = (PLOT_R - PLOT_L) / float(n)
    centers = [PLOT_L + step * (i + 0.5) for i in range(n)]
    def y_of(v):
        return PLOT_B + v / VMAX * (PLOT_T - PLOT_B)
    for q in range(4):
        i0 = q * 3
        i1 = q * 3 + 2
        x0 = centers[i0] - step * 0.35
        x1 = centers[i1] + step * 0.35
        yt = y_of(Q_TOTALS[q])
        ax.add_patch(plt.Rectangle(
            (x0, PLOT_B), x1 - x0, yt - PLOT_B,
            facecolor=Q_BG_COLORS[q], edgecolor="none", alpha=0.55, zorder=2,
        ))
        ax.text(
            (x0 + x1) / 2.0, yt + 12.0, "%d" % Q_TOTALS[q],
            fontproperties=fp(10.5, False, "num"),
            color=TXT, ha="center", va="bottom", zorder=8,
        )
    bar_w = step * 0.22
    for i in range(n):
        q = i // 3
        h = y_of(VALUES[i]) - PLOT_B
        x0 = centers[i] - bar_w / 2.0
        ax.add_patch(plt.Rectangle(
            (x0, PLOT_B), bar_w, h,
            facecolor=Q_BAR_COLORS[q], edgecolor="none", zorder=4,
        ))
        ax.text(
            centers[i], PLOT_B + h + 8.0, "%d" % VALUES[i],
            fontproperties=fp(9.0, False, "num"),
            color=TXT, ha="center", va="bottom", zorder=8,
        )
        ax.text(
            centers[i], PLOT_B - 18.0, MONTHS[i],
            fontproperties=fp(10.0, False, "ea"),
            color="#D9D9D9", ha="center", va="top", zorder=8,
        )
    ax.plot([PLOT_L, PLOT_R], [PLOT_B, PLOT_B], color=AXIS_LINE, linewidth=0.8, zorder=1)
    fig.text(
        0.06, 1.0 - (28.0) / H, "2021年各月化妆品销量走势",
        fontproperties=fp(20.0, True, "ea"),
        color=TXT, ha="left", va="top", zorder=30,
    )
    fig.text(
        0.06, 1.0 - (76.0) / H, "2021年第三季度销量最多9673，9月单月销量最大3621",
        fontproperties=fp(11.5, False, "ea"),
        color=TXT, ha="left", va="top", zorder=30,
    )
    fig.text(
        0.06, 1.0 - (495.0) / H, "*注：数据来源于公司销售系统，统计日期截至2021.12.31",
        fontproperties=fp(9.0, False, "ea"),
        color=NOTE, ha="left", va="top", zorder=30,
    )
    return fig
if __name__ == "__main__":
    f = draw()
    out = os.path.join(OUT, "chart28_复合柱形图.png")
    f.savefig(out, dpi=DPI, facecolor=BG)
    try:
        from IPython import get_ipython
        if get_ipython() is not None:
            from IPython.display import display
            display(f)
    except Exception:
        pass
    plt.close(f)
    print("已输出：", out)
