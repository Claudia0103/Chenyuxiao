import os
import numpy as np
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
from matplotlib import font_manager as fm
from matplotlib.patches import Wedge
try:
    HERE = os.path.dirname(os.path.abspath(__file__))
except NameError:
    HERE = os.path.abspath(os.getcwd())
OUT = os.path.join(HERE, "png")
os.makedirs(OUT, exist_ok=True)
DPI = 120.0
EM_PT = DPI / 72.0
SIZE_PX = [640, 560]
BG = "#1A1E43"
TXT = "#FFFFFF"
BLUE = "#1565A8"
CYAN = "#1F9B96"
YELLOW = "#F0B03F"
RED = "#E84E6A"
VALUE = 76
V_MIN = 50
V_MAX = 150
A_START = 225.0
A_SPAN = 270.0
_FONT_FILES = {
    "ea": {"n": "C:/Windows/Fonts/msyh.ttc", "b": "C:/Windows/Fonts/msyhbd.ttc"},
    "cal": {"n": "C:/Windows/Fonts/calibri.ttf", "b": "C:/Windows/Fonts/calibrib.ttf"},
    "num": {"n": "C:/Windows/Fonts/arial.ttf", "b": "C:/Windows/Fonts/arialbd.ttf"},
}
for _d in _FONT_FILES.values():
    for _p in _d.values():
        try:
            fm.fontManager.addfont(_p)
        except Exception:
            pass
_FP_CACHE = {}
def fp(sz, bold=False, fam="ea"):
    key = (round(float(sz), 2), bool(bold), fam)
    p = _FP_CACHE.get(key)
    if p is None:
        p = fm.FontProperties(fname=_FONT_FILES[fam]["b" if bold else "n"])
        p.set_size(sz)
        _FP_CACHE[key] = p
    return p
def v2deg(v):
    t = (v - V_MIN) / (V_MAX - V_MIN)
    a = A_START - t * A_SPAN
    return a
def draw():
    W, H = SIZE_PX
    fig = plt.figure(figsize=(W / DPI, H / DPI), dpi=DPI)
    fig.patch.set_facecolor(BG)
    ax = fig.add_axes([0, 0, 1, 1])
    ax.set_xlim(-1.4, 1.4)
    ax.set_ylim(-1.6, 1.2)
    ax.set_aspect("equal")
    ax.set_facecolor(BG)
    for side in ("top", "right", "bottom", "left"):
        ax.spines[side].set_visible(False)
    ax.set_xticks([])
    ax.set_yticks([])
    cx, cy = 0.0, -0.25
    r_out = 0.90
    r_out_in = 0.82
    r_seg_out = r_out_in
    r_seg_in = 0.71
    r_num = 0.62
    ax.add_patch(Wedge(
        (cx, cy), r_out, 0, 360,
        width=r_out - r_out_in, facecolor=BLUE, edgecolor="none", zorder=2,
    ))
    segs = [
        (50, 80, CYAN),
        (80, 120, YELLOW),
        (120, 150, RED),
    ]
    for v0, v1, col in segs:
        a0 = v2deg(v1)
        a1 = v2deg(v0)
        ax.add_patch(Wedge(
            (cx, cy), r_seg_out, a0, a1,
            width=r_seg_out - r_seg_in, facecolor=col,
            edgecolor="none", zorder=3,
        ))
    for v in range(V_MIN, V_MAX + 1, 10):
        a = v2deg(v)
        rad = np.deg2rad(a)
        x0 = cx + r_seg_in * np.cos(rad)
        y0 = cy + r_seg_in * np.sin(rad)
        x1 = cx + r_seg_out * np.cos(rad)
        y1 = cy + r_seg_out * np.sin(rad)
        ax.plot([x0, x1], [y0, y1], color="#1A1E43", linewidth=1.0, zorder=4)
    for v in range(V_MIN, V_MAX + 1, 10):
        a = v2deg(v)
        rad = np.deg2rad(a)
        x0 = cx + r_out_in * np.cos(rad)
        y0 = cy + r_out_in * np.sin(rad)
        x1 = cx + r_out * np.cos(rad)
        y1 = cy + r_out * np.sin(rad)
        ax.plot([x0, x1], [y0, y1], color="#1A1E43", linewidth=1.0, zorder=5)
    for v in range(V_MIN, V_MAX + 1, 10):
        a = v2deg(v)
        rad = np.deg2rad(a)
        xt = cx + r_num * np.cos(rad)
        yt = cy + r_num * np.sin(rad)
        ax.text(
            xt, yt, "%d" % v,
            fontproperties=fp(11.0, True, "num"),
            color=TXT, ha="center", va="center", zorder=8,
        )
    a_val = v2deg(VALUE)
    rad = np.deg2rad(a_val)
    nx = cx + r_seg_in * 0.9 * np.cos(rad)
    ny = cy + r_seg_in * 0.9 * np.sin(rad)
    ax.plot([cx, nx], [cy, ny], color="#FFFFFF", linewidth=1.4, zorder=10)
    ax.text(
        cx, cy - 0.55, "%d" % VALUE,
        fontproperties=fp(38.0, True, "num"),
        color=TXT, ha="center", va="center", zorder=12,
    )
    fig.text(
        0.5, 1.0 - (40.0) / H, "2022年6月29公司整体运营指数良好",
        fontproperties=fp(20.0, True, "ea"),
        color=TXT, ha="center", va="top", zorder=30,
    )
    return fig
if __name__ == "__main__":
    f = draw()
    out = os.path.join(OUT, "chart22_仪表盘图.png")
    f.savefig(out, dpi=DPI, facecolor=BG)
    try:
        from IPython import get_ipython
        if get_ipython() is not None:
            from IPython.display import display
            display(f)
    except Exception:
        pass
    plt.close(f)
    print("已输出：", out)
