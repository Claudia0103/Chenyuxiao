def plot_domain_map():
    fig, ax = plt.subplots(figsize=(11, 7))
    ax.set_xlim(0, 14)
    ax.set_ylim(0, 10)
    ax.axis("off")

    def card(x, y, w, h, title, ents, color, opt=False):
        b = FancyBboxPatch((x, y), w, h,
                           boxstyle="round,pad=0.05,rounding_size=0.12",
                           fc="#f4f6fc", ec=color, lw=1.4)
        ax.add_patch(b)
        ax.text(x + w / 2, y + h - 0.32, title + ("（可选）" if opt else ""),
                ha="center", fontsize=10.5, color=color, fontweight="bold")
        ax.text(x + w / 2, y + h / 2 - 0.18, ents, ha="center", va="center",
                fontsize=8.6, color=INK)

    # 三个核心主题域（人/货/场）大字突出
    card(1.0, 6.3, 3.4, 2.0, "消费者（人）", "会员 Member\n访客 Visitor", "#b02a37")
    card(5.3, 6.3, 3.4, 2.0, "商品（货）", "商品 Merchandise\n货品 Product", "#1a6f3c")
    card(9.6, 6.3, 3.4, 2.0, "商家（场）", "门店 Store\n站点 Site", ORANGE)

    # 其余主题域 2 行
    rest = [(t, e, o) for (t, g, e, o) in DOMAINS if g == "支撑"]
    for i, (t, e, o) in enumerate(rest[:5]):
        card(0.9 + i * 2.55, 3.2, 2.35, 1.7, t, e, BLUE, o)
    for i, (t, e, o) in enumerate(rest[5:]):
        card(2.15 + i * 2.55, 0.9, 2.35, 1.7, t, e, BLUE, o)

    ax.text(7.0, 9.4, "零售行业主题域模型：核心主题域为“人、货、场”（白皮书 P06）",
            ha="center", fontsize=13, fontweight="bold")
    ax.text(7.0, 8.75, "主题域划分没有客观原则，依据数据建模师的行业经验与业务理解，"
                       "核心主题域为“人”“货”“场”",
            ha="center", fontsize=9, color=GREY)
    plt.tight_layout()
    p = os.path.join(CHART_DIR, "12.2_主题域全景图.png")
    plt.savefig(p, dpi=130, bbox_inches="tight")
    plt.close()
    print(f"[图] 已保存：{p}")
    return p

# 2) ER 图例（白皮书 P07 5.2 的元元素类型表）
# ============================================================
def _obj(ax, x, y, w=1.5, h=0.62, label="", dashed=False):
    """业务对象实体：直角矩形（泛化实体用虚线）"""
    r = Rectangle((x - w / 2, y - h / 2), w, h, fc="white", ec=INK, lw=1.2,
                  ls="--" if dashed else "-")
    ax.add_patch(r)
    ax.text(x, y, label, ha="center", va="center", fontsize=9, color=INK)


def _act(ax, x, y, w=1.5, h=0.62, label="", dashed=False):
    """业务活动实体：平行四边形（泛化活动用虚线）"""
    s = 0.22
    poly = Polygon([(x - w / 2 + s, y + h / 2), (x + w / 2 + s, y + h / 2),
                    (x + w / 2 - s, y - h / 2), (x - w / 2 - s, y - h / 2)],
                   closed=True, fc="#eef1fa", ec=INK, lw=1.2,
                   ls="--" if dashed else "-")
    ax.add_patch(poly)
    ax.text(x, y, label, ha="center", va="center", fontsize=9, color=INK)


def _one(ax, x, y):
    ax.plot([x - 0.09, x + 0.09], [y, y], color=INK, lw=1.4)
    ax.plot([x, x], [y - 0.09, y + 0.09], color=INK, lw=1.4)  # 十字=强制1


def _many(ax, x, y):
    ax.plot([x, x - 0.16], [y, y + 0.1], color=INK, lw=1.2)
    ax.plot([x, x - 0.16], [y, y - 0.1], color=INK, lw=1.2)
    ax.plot([x, x - 0.12], [y, y], color=INK, lw=1.2)  # 鸟脚=多


def plot_er_legend():
    fig, ax = plt.subplots(figsize=(10.5, 5.2))
    ax.set_xlim(0, 14)
    ax.set_ylim(0, 7.4)
    ax.axis("off")
    ax.text(7.0, 7.0, "概念模型 ER 图例（白皮书 P07，5.2 元元素类型）",
            ha="center", fontsize=13, fontweight="bold")

    # 左列：实体类型
    _obj(ax, 2.0, 6.0, label="业务对象实体")
    _act(ax, 2.0, 5.0, label="业务活动实体")
    _obj(ax, 2.0, 4.0, label="泛化实体", dashed=True)
    ax.text(4.0, 4.0, "虚线矩形：概念上并不存在真实实例的抽象实体\n"
                      "（如“商家”：系统中并无真实实体表）",
            fontsize=8.5, va="center", color=GREY)

    # 右列：关系类型
    y = 6.0
    ax.plot([8.3, 11.2], [y, y], color=INK, lw=1.2)
    _one(ax, 8.3, y); _one(ax, 11.2, y)
    _obj(ax, 7.0, y, w=1.2, h=0.5, label="实体A")
    _obj(ax, 12.5, y, w=1.2, h=0.5, label="实体B")
    ax.text(13.4, y, "一对一引用", fontsize=9.5, va="center")

    y = 5.0
    ax.plot([8.3, 11.2], [y, y], color=INK, lw=1.2)
    _one(ax, 8.3, y); _many(ax, 11.2, y)
    _obj(ax, 7.0, y, w=1.2, h=0.5, label="实体A")
    _obj(ax, 12.5, y, w=1.2, h=0.5, label="实体B")
    ax.text(13.4, y, "一对多引用", fontsize=9.5, va="center")

    y = 4.0
    ax.plot([8.3, 11.2], [y, y], color=INK, lw=1.2)
    _many(ax, 8.3, y); _many(ax, 11.2, y)
    _obj(ax, 7.0, y, w=1.2, h=0.5, label="实体A")
    _obj(ax, 12.5, y, w=1.2, h=0.5, label="实体B")
    ax.text(13.4, y, "多对多引用", fontsize=9.5, va="center")

    y = 3.0
    ax.plot([8.3, 10.9], [y, y], color=INK, lw=1.2)
    ax.add_patch(FancyArrowPatch((10.9, y), (11.4, y), arrowstyle="-|>",
                                 mutation_scale=12, color=INK, lw=1.2))
    _obj(ax, 7.0, y, w=1.2, h=0.5, label="买家")
    _obj(ax, 12.5, y, w=1.2, h=0.5, label="用户")
    ax.text(13.4, y, "继承（买家/会员 继承 用户）", fontsize=9.5, va="center")

    # 底部注解
    ax.text(7.0, 1.8, "实体关系有两种：\n"
                      "a. 引用关系：一个实体是另一个实体的属性（如订单引用了买家、卖家、商品），\n"
                      "    技术上就是 SQL 中的“关联”，分一对一/一对多/多对多；\n"
                      "b. 继承关系：A 从属于 B 且更细化具体（买家=有过交易的用户，会员=参与会员项目的用户）。",
            fontsize=9, color=INK, ha="center", va="center")
    plt.tight_layout()
    p = os.path.join(CHART_DIR, "12.2_ER图例.png")
    plt.savefig(p, dpi=130, bbox_inches="tight")
    plt.close()
    print(f"[图] 已保存：{p}")
    return p

# 3) 三个典型主题域的概念模型 ER 图（简化重绘）
# ============================================================
def _link(ax, x1, y1, x2, y2, a="1", b="1"):
    """引用关系连线：a/b 取 '1' 或 'N'"""
    ax.plot([x1, x2], [y1, y2], color=INK, lw=1.1)
    if a == "1":
        _one(ax, x1, y1)
    else:
        _many(ax, x1, y1)
    if b == "1":
        _one(ax, x2, y2)
    else:
        _many(ax, x2, y2)


def plot_er_common():
    """公共域（白皮书 5.3）：地理行政区划-地址-员工-部门；用户/收货地址；
    供应商/仓库/门店 引用 地址"""
    fig, ax = plt.subplots(figsize=(10.5, 6.4))
    ax.set_xlim(0, 14)
    ax.set_ylim(0, 9)
    ax.axis("off")
    ax.text(7.0, 8.6, "公共域概念模型 ER 图（白皮书 5.3，简化版）",
            ha="center", fontsize=13, fontweight="bold")

    # 域边界
    ax.add_patch(Rectangle((0.5, 1.1), 9.4, 6.6, fc="#f6f7fb", ec=GREY,
                           lw=1.1, ls="--"))
    ax.text(1.0, 7.35, "公共域", fontsize=10.5, color=GREY)

    _obj(ax, 2.3, 6.2, label="收货地址")
    _obj(ax, 6.2, 6.2, label="用户")
    _link(ax, 3.05, 6.2, 5.45, 6.2, a="N", b="1")   # 收货地址 N:1 用户
    _obj(ax, 2.6, 4.6, label="地理行政区划")
    _obj(ax, 6.2, 4.6, label="地址")
    _obj(ax, 2.6, 2.6, label="部门")
    _obj(ax, 6.2, 2.6, label="员工")
    _link(ax, 2.6, 5.05, 2.6, 3.35)  # 区划-部门(示意:归属)
    ax.text(2.2, 4.2, "归属", fontsize=8, color=GREY, rotation=90)
    _link(ax, 3.35, 4.6, 5.45, 4.6)  # 区划 1:N 地址
    _link(ax, 6.2, 5.05, 6.2, 3.35)  # 地址 1:N 员工
    _link(ax, 3.35, 2.6, 5.45, 2.6, a="1", b="N")  # 部门 1:N 员工

    # 域外：供应商/仓库/门店 引用 地址（N:1）
    _obj(ax, 12.2, 6.4, label="供应商")
    _obj(ax, 12.2, 4.6, label="仓库")
    _obj(ax, 12.2, 2.8, label="门店")
    ax.plot([7.0, 10.6], [4.6, 4.6], color=INK, lw=1.1)      # 地址右出主干
    for yy in (6.4, 4.6, 2.8):
        ax.plot([10.6, 11.45], [4.6, yy], color=INK, lw=1.1)
        _many(ax, 11.45, yy)
    _one(ax, 7.0, 4.6)
    ax.text(11.0, 1.95, "供应商/仓库/门店 引用地址（N:1）",
            fontsize=8.5, color=GREY)

    ax.text(7.0, 0.5, "地址=某个人或商业实体(门店、仓库)的具体地理位置；"
                      "地理行政区划与员工均为树状层级结构",
            ha="center", fontsize=9, color=GREY)
    plt.tight_layout()
    p = os.path.join(CHART_DIR, "12.2_公共域ER图.png")
    plt.savefig(p, dpi=130, bbox_inches="tight")
    plt.close()
    print(f"[图] 已保存：{p}")
    return p
