from matplotlib.patches import FancyBboxPatch, FancyArrowPatch

try:
    BASE_DIR = os.path.dirname(os.path.abspath(__file__))

# 统一配色（与白皮书深蓝/浅紫风格呼应，尽量少用色）

# ============================================================
# 1) 分层数据模型：行业数据模型的 6 层组成（白皮书 P04 左栏）
# ============================================================
MODEL_LAYERS = [
    ("1. 行业业务流程", "行业的市场角色与要素流动（本白皮书第 3 章）"),
    ("2. 行业主题域模型", "宏观分析领域划分，如零售的“人、货、场”（第 4 章）"),
    ("3. 概念模型", "主题域内增加实体以及实体之间的关系（第 5 章）"),
    ("4. 逻辑模型", "概念模型基础上，增加每个实体的属性及属性约束（第 6 章）"),
    ("5. 业务分析模型", "行业常用分析视角：原子指标 + 业务限定 + 统计粒度/周期\n→ 派生指标（第 7 章）"),
    ("6. 行业应用场景", "可选。基于以上模型的数据驱动场景方案，如推荐系统、用户营销"),
]


def plot_layered_model():
    fig, ax = plt.subplots(figsize=(9.5, 7))
    ax.set_xlim(0, 10)
    ax.set_ylim(0, 10)
    ax.axis("off")

    n = len(MODEL_LAYERS)
    top, bottom = 9.4, 0.6
    h = (top - bottom) / n - 0.25
    for i, (t, d) in enumerate(MODEL_LAYERS):
        y = top - i * ((top - bottom) / n) - h / 2
        # 每层一个圆角横条：左侧层名，右侧说明
        box = FancyBboxPatch((0.6, y - h / 2), 8.8, h,
                             boxstyle="round,pad=0.06,rounding_size=0.14",
                             fc=LIGHT if i % 2 == 0 else "#eef1fa",
                             ec=BLUE, lw=1.2)
        ax.add_patch(box)
        ax.text(1.0, y, t, ha="left", va="center", fontsize=11.5,
                fontweight="bold", color=BLUE)
        ax.text(4.15, y, d, ha="left", va="center", fontsize=8.6, color=GREY)
        if i < n - 1:
            y_next = top - (i + 1) * ((top - bottom) / n) - h / 2
            ax.add_patch(FancyArrowPatch((5.0, y - h / 2 - 0.02),
                                         (5.0, y_next + h / 2 + 0.02),
                                         arrowstyle="-|>", mutation_scale=13,
                                         color=BLUE, lw=1.2))

    ax.set_title("阿里云数据中台行业数据模型：分层数据模型（白皮书 P04）",
                 fontsize=13, fontweight="bold")
    plt.tight_layout()
    p = os.path.join(CHART_DIR, "12.1_分层数据模型.png")
    plt.savefig(p, dpi=130, bbox_inches="tight")
    plt.close()
    print(f"[图] 已保存：{p}")
    return p


# ============================================================
# 2) 业务分析模型子图（白皮书 P04 右上图）
#    原子指标 + 业务限定 + 统计粒度（+统计周期）→ 派生指标
#    主题域模型 --细化--> 概念模型 --细化--> 逻辑模型
# ============================================================
def plot_analysis_model():
    fig, ax = plt.subplots(figsize=(10.5, 6.2))
    ax.set_xlim(0, 14)
    ax.set_ylim(0, 10)
    ax.axis("off")

    def box(x, y, w, h, text, fc, ec=BLUE, fs=10, tc="white"):
        b = FancyBboxPatch((x - w / 2, y - h / 2), w, h,
                           boxstyle="round,pad=0.05,rounding_size=0.12",
                           fc=fc, ec=ec, lw=1.2)
        ax.add_patch(b)
        ax.text(x, y, text, ha="center", va="center", fontsize=fs, color=tc)

    def arrow(x1, y1, x2, y2, label=None):
        ax.add_patch(FancyArrowPatch((x1, y1), (x2, y2), arrowstyle="-|>",
                                     mutation_scale=13, color=GREY, lw=1.2))
        if label:
            ax.text((x1 + x2) / 2, (y1 + y2) / 2 + 0.18, label,
                    ha="center", fontsize=9, color=GREY)

    # ---- 顶部：派生指标的构成 ----
    ax.text(7, 9.55, "业务分析模型：派生指标的构成", ha="center",
            fontsize=11.5, fontweight="bold", color=BLUE)
    y0 = 8.3
    box(1.7, y0, 2.3, 1.0, "统计周期", PURPLE)
    box(4.4, y0, 2.3, 1.0, "原子指标", BLUE)
    box(7.1, y0, 2.3, 1.0, "业务限定", BLUE)
    box(9.8, y0, 2.3, 1.0, "统计粒度", BLUE)
    ax.text(3.05, y0 + 0.65, "+", fontsize=13, ha="center", color=GREY)
    ax.text(5.75, y0 + 0.65, "+", fontsize=13, ha="center", color=GREY)
    ax.text(8.45, y0 + 0.65, "+", fontsize=13, ha="center", color=GREY)
    box(12.6, y0, 2.3, 1.0, "派生指标", "#1f3f77")
    arrow(10.95, y0, 11.45, y0)

    # ---- 中部：三条建模路径 ----
    y1 = 5.6
    cols = [("主题域模型", ["主题域1", "主题域2", "主题域n"], LIGHT),
            ("概念模型", ["维度", "业务过程"], "#c9cdf0"),
            ("逻辑模型", ["维度逻辑表", "事实逻辑表"], "#aeb5e6")]
    for ci, (title, items, fc) in enumerate(cols):
        cx = 2.6 + ci * 4.4
        ax.text(cx, y1 + 1.05, title, ha="center", fontsize=11,
                fontweight="bold", color=BLUE)
        for ii, it in enumerate(items):
            box(cx, y1 - ii * 1.35, 2.6, 0.95, it, fc, tc="#22242e", fs=9.5)
        if ci < 2:
            ax.text(cx + 2.2, y1 - 1.3, "细化", fontsize=9, color=GREY,
                    rotation=90, ha="center", va="center")
            arrow(cx + 2.2, y1 + 0.6, cx + 2.2, y1 - 0.2)
    # 底部箭头：分析视角落到模型上
    arrow(7.0, 7.75, 7.0, 6.9)
    ax.text(7.25, 7.3, "每个分析视角对应一套模型", fontsize=8.5, color=GREY)

    ax.set_title("业务分析视角自上而下落到主题域/概念/逻辑模型（白皮书 P04 右图）",
                 fontsize=12.5, fontweight="bold")
    plt.tight_layout()
    p = os.path.join(CHART_DIR, "12.1_业务分析模型构成.png")
    plt.savefig(p, dpi=130, bbox_inches="tight")
    plt.close()
    print(f"[图] 已保存：{p}")
    return p

ONEMODEL_EXAMPLE = {
    "派生指标": "最近一天每个门店男性成交客户数",
    "原子指标": "成交客户数（最基本的统计计数，没有任何其他约束条件）",
    "统计周期": "最近一天（时间范围约束）",
    "业务限定": "男性 / 需有成交记录（业务范围约束）",
    "统计粒度": "门店（统计的颗粒度）",
}


def plot_onemodel():
    fig, ax = plt.subplots(figsize=(10.5, 5.6))
    ax.set_xlim(0, 14)
    ax.set_ylim(0, 8)
    ax.axis("off")

    # 顶部：派生指标完整名
    b = FancyBboxPatch((2.2, 6.5), 9.6, 1.0,
                       boxstyle="round,pad=0.06,rounding_size=0.15",
                       fc="#1f3f77", ec="none")
    ax.add_patch(b)
    ax.text(7.0, 7.0, f"派生指标：{ONEMODEL_EXAMPLE['派生指标']}",
            ha="center", va="center", fontsize=13, color="white",
            fontweight="bold")

    # 四个要素框
    elems = [("统计周期", "最近一天"), ("原子指标", "成交客户数"),
             ("业务限定", "男性（需有\n成交记录）"), ("统计粒度", "门店")]
    xs = [2.0, 5.3, 8.6, 11.9]
    for (t, v), x in zip(elems, xs):
        bb = FancyBboxPatch((x - 1.4, 3.1), 2.8, 1.9,
                            boxstyle="round,pad=0.06,rounding_size=0.12",
                            fc=BLUE if t != "原子指标" else PURPLE, ec="none")
        ax.add_patch(bb)
        ax.text(x, 4.5, t, ha="center", va="center", fontsize=12,
                color="white", fontweight="bold")
        ax.text(x, 3.72, v, ha="center", va="center", fontsize=9.5,
                color="white")
        # 向上汇聚箭头
        ax.add_patch(FancyArrowPatch((x, 5.05), (x * 0.55 + 7.0 * 0.45, 6.42),
                                     arrowstyle="-|>", mutation_scale=11,
                                     color=GREY, lw=1.1,
                                     connectionstyle="arc3,rad=0"))

    # 底部说明
    ax.text(7.0, 1.9,
            "OneModel 把最终用于分析与应用的指标定义为“派生指标”，"
            "并拆解为 4 个元素：",
            ha="center", fontsize=10, color="#22242e")
    ax.text(7.0, 1.15,
            "原子指标（最基本统计）+ 统计周期（时间范围）+ "
            "业务限定（业务范围）+ 统计粒度（统计颗粒度）",
            ha="center", fontsize=10, color=BLUE, fontweight="bold")
    ax.text(7.0, 0.45,
            "企业规模越大、指标越多，OneModel 保障指标全层面统一业务认知、避免重复建设",
            ha="center", fontsize=8.5, color=GREY)

    ax.set_title("OneModel：派生指标 = 原子指标 + 统计周期 + 业务限定 + 统计粒度",
                 fontsize=12.5, fontweight="bold")
    plt.tight_layout()
    p = os.path.join(CHART_DIR, "12.1_OneModel四要素.png")
    plt.savefig(p, dpi=130, bbox_inches="tight")
    plt.close()
    print(f"[图] 已保存：{p}")

# 4) 可运行的 OneModel 派生指标注册器
#    四要素组合 → 规范命名 → 查重登记（白皮书 2.2 的方法落地）
# ============================================================
class OneModelRegistry:
    """派生指标注册器：登记时按四要素生成规范名，重名/重复定义直接报错。"""

    def __init__(self):
        self._by_name = {}
        self._by_def = {}

    def register(self, name, atom, period, limit, grain, note=""):
        # 1) 命名唯一性：派生指标名不能重复
        if name in self._by_name:
            raise ValueError(f"派生指标重名：{name}")
        # 2) 定义唯一性：同一组四要素不允许重复建设（OneModel 的核心目的）
        key = (atom, period, limit, grain)
        if key in self._by_def:
            old = self._by_def[key]
            raise ValueError(
                f"四要素重复定义：{old['指标名']} 已是 "
                f"{atom}+{period}+{limit}+{grain}，无需重复建设")
        rec = {"指标名": name, "原子指标": atom, "统计周期": period,
               "业务限定": limit, "统计粒度": grain, "说明": note}
        self._by_name[name] = rec
        self._by_def[key] = rec
        return rec

    def to_frame(self):
        return pd.DataFrame(list(self._by_name.values()))


def demo_registry():
    reg = OneModelRegistry()
    # 演示指标取自白皮书第 7 章 AIPL / 全局 KPI 的真实派生指标
    reg.register("最近一天每个门店男性成交客户数", "成交客户数",
                 "最近一天", "男性", "门店", "白皮书 P04 例")
    reg.register("最近XX天月广告触达次数", "次数(触达)",
                 "最近XX天/月", "广告触达", "用户(消费者)", "AIPL-A")
    reg.register("最近XX天月访问次数", "次数(访问)",
                 "最近XX天/月", "无", "用户(消费者)", "AIPL-I")
    reg.register("最近XX天月购买次数", "次数(购买)",
                 "最近XX天/月", "无", "用户(消费者)", "AIPL-P")
    reg.register("最近XX天月分享次数", "次数(分享)",
                 "最近XX天/月", "无", "用户(消费者)", "AIPL-L")
    reg.register("最近一次购买距今天数(R)", "天数(最近购买)",
                 "历史截止当前", "无", "客户(消费者)", "RFM-R")
    reg.register("累计购买金额", "sum(金额)",
                 "历史截止当前", "无", "客户(消费者)", "RFM-M")
    reg.register("最近一天/月成交客户数", "客户计数",
                 "最近一天/月", "全局", "全局性(跨年龄/性别等)", "全局 KPI")

    df = reg.to_frame()
    print("=" * 76)
    print("OneModel 派生指标注册表（演示数据，取自白皮书第 7 章真实指标）")
    print("=" * 76)
    print(df[["指标名", "原子指标", "统计周期", "统计粒度"]].to_string(index=False))
    print(f"\n共登记 {len(df)} 个派生指标。")

    # 查重演示：重复命名 → 报错；同四要素换名 → 报错
    try:
        reg.register("最近XX天月购买次数", "次数(购买)", "最近XX天/月",
                     "无", "用户(消费者)")
    except ValueError as e:
        print(f"\n[查重演示-命名冲突] {e}")
    try:
        reg.register("换个名字的购买次数", "次数(购买)", "最近XX天/月",
                     "无", "用户(消费者)")
    except ValueError as e:
        print(f"[查重演示-重复建设] {e}")
    return df
