# 把中文「计算规则」翻译成能执行的公式。
# 每项 = (依赖的指标名列表, 计算函数, 单位)
DERIVED = {
    "30天内新客进件转化率": (["30天内新客申请贷款数", "注册用户数"],
                       lambda v: v["30天内新客申请贷款数"] / v["注册用户数"], "%"),
    "审核通过率": (["审核通过用户数", "进件用户数"],
              lambda v: v["审核通过用户数"] / v["进件用户数"], "%"),
    "机器审核通过率": (["系统自动审核通过用户数", "进件用户数"],
                lambda v: v["系统自动审核通过用户数"] / v["进件用户数"], "%"),
    "人工审核通过率": (["人工审核通过用户数", "进件用户数"],
                lambda v: v["人工审核通过用户数"] / v["进件用户数"], "%"),
    "放款转化率": (["放款用户数", "审核通过用户数"],
              lambda v: v["放款用户数"] / v["审核通过用户数"], "%"),
    "放款件均": (["放款金额", "放款用户数"],
             lambda v: v["放款金额"] / v["放款用户数"], "元"),
    "30天内复贷转化率": (["合同结清30天内再次进件数", "合同结清数"],
                  lambda v: v["合同结清30天内再次进件数"] / v["合同结清数"], "%"),

def save(fig, name):
    path = os.path.join(CHART_DIR, name)
    fig.savefig(path, dpi=150, bbox_inches="tight")
    plt.close(fig)
    print("图表已存:", path)


def node(ax, x, y, w, h, text, fc, fontsize=9, tc="white"):
    """画一个圆角框（自己写，不依赖 graphviz / networkx）。"""
    ax.add_patch(FancyBboxPatch((x, y), w, h, boxstyle="round,pad=0.01,rounding_size=0.012",
                                facecolor=fc, edgecolor="white", linewidth=1))
    ax.text(x + w / 2, y + h / 2, text, ha="center", va="center",
            fontsize=fontsize, color=tc, wrap=True)


def link(ax, x1, y1, x2, y2, color="#9AA5B1"):
    ax.add_patch(FancyArrowPatch((x1, y1), (x2, y2), arrowstyle="-",
                                 linewidth=1, color=color,
                                 connectionstyle="arc3,rad=0"))


def chart_osm():
    """71a：OSM 三层结构，O 一层、S 五个策略、M 每个策略下的度量。"""
    fig, ax = plt.subplots(figsize=(13, 7))
    ax.set_xlim(0, 1)
    ax.set_ylim(0, 1)
    ax.axis("off")
    ax.set_title("Step1  OSM 模型：目标 → 策略 → 度量（课件第 20 页 GMV 示例）", fontsize=14)

    node(ax, 0.36, 0.87, 0.28, 0.075, "O  " + OSM["目标 O"], "#1F3864", 12)

    keys = list(OSM["策略 S"].keys())
    w = 0.176
    gap = 0.012
    x0 = 0.028
    for i, k in enumerate(keys):
        x = x0 + i * (w + gap)
        node(ax, x, 0.70, w, 0.075, "S  " + k.split()[0], "#2F5597", 10)
        link(ax, 0.50, 0.87, x + w / 2, 0.775)
        # 策略下的行动（灰框）和度量（浅蓝框）
        acts = OSM["策略 S"][k]
        mets = OSM["度量 M"][k]
        for j, t in enumerate(acts):
            yy = 0.60 - j * 0.048
            node(ax, x + 0.005, yy, w - 0.01, 0.038, t, "#D6DEEB", 8, "#333333")
            link(ax, x + w / 2, 0.70, x + w / 2, yy + 0.038, "#C7D0DE")
        base = 0.60 - len(acts) * 0.048
        for j, t in enumerate(mets):
            yy = base - 0.055 - j * 0.048
            node(ax, x + 0.005, yy, w - 0.01, 0.038, "M  " + t, "#9DC3E6", 8, "#14293F")
            link(ax, x + w / 2, base + 0.038, x + w / 2, yy + 0.038, "#C7D0DE")
    ax.text(0.5, 0.02, "灰框 = 行动策略 S     蓝框 = 度量指标 M", ha="center",
            fontsize=9, color="#666666")
    save(fig, "71a_OSM指标拆解.png")


def compute_metrics(values):
    """按「关联指标」依赖做拓扑排序，先算基础再算派生。"""
    all_names = set(values) | set(DERIVED)
    deps = {n: set() for n in all_names}
    for name, (d, _, _) in DERIVED.items():
        deps[name] = set(d)

    order, pending = [], set(all_names)
    while pending:
        ready = sorted(n for n in pending if not (deps[n] & pending))
        if not ready:
            raise RuntimeError("指标依赖成环，剩: " + str(pending))
        order.extend(ready)
        pending -= set(ready)

    result = dict(values)
    for n in order:
        if n in DERIVED:
            result[n] = DERIVED[n][1](result)
    return result, order


def chart_dependency(result):
    """71b：基础指标 → 派生指标 的依赖图。"""
    basics = [k for k in BASE_VALUES if k not in DERIVED]
    derived = list(DERIVED.keys())

    fig, ax = plt.subplots(figsize=(11, 10))
    ax.set_xlim(0, 1)
    ax.set_ylim(0, 1)
    ax.axis("off")
    ax.set_title("Step2  指标依赖图：基础指标（左） → 派生指标（右）", fontsize=14)

    hb = 0.072
    yb = [0.90 - i * (hb + 0.018) for i in range(len(basics))]
    hd = 0.072
    yd = [0.90 - i * (hd + 0.045) for i in range(len(derived))]

    for i, (k, y) in enumerate(zip(basics, yb)):
        node(ax, 0.02, y, 0.30, hb, k, "#E8EEF7", 9, "#1F3864")
    for i, (k, y) in enumerate(zip(derived, yd)):
        node(ax, 0.62, y, 0.34, hd, k, "#F58518", 9)
        v = result[k]
        unit = DERIVED[k][2]
        txt = "{:.2f}%".format(v * 100) if unit == "%" else "{:,.0f} {}".format(v, unit)
        ax.text(0.985, y + hd / 2, txt, va="center", fontsize=9, color="#C00000")
        for d in DERIVED[k][0]:
            if d in basics:
                j = basics.index(d)
                link(ax, 0.32, yb[j] + hb / 2, 0.62, y + hd / 2)
    save(fig, "71b_指标依赖图.png")


def chart_dimension():
    """71c：模板的「维度」列落到实操 —— 同一指标按渠道、按新老客下钻。"""
    df = pd.DataFrame(CHANNEL_DATA, columns=["渠道", "注册用户数", "进件用户数", "新老客"])
    df["进件转化率"] = df["进件用户数"] / df["注册用户数"]
    by_type = df.groupby("新老客")[["注册用户数", "进件用户数"]].sum()
    by_type["进件转化率"] = by_type["进件用户数"] / by_type["注册用户数"]

    fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(11, 4.6))
    bars = ax1.bar(df["渠道"], df["进件转化率"] * 100, color="#4C78A8", width=0.6)
    for b, v in zip(bars, df["进件转化率"] * 100):
        ax1.text(b.get_x() + b.get_width() / 2, v + 0.5, "{:.1f}%".format(v),
                 ha="center", fontsize=10)
    ax1.set_ylim(0, max(df["进件转化率"] * 100) * 1.25)
    ax1.set_ylabel("进件转化率（%）")
    ax1.set_title("按「渠道」维度下钻")
    ax1.grid(axis="y", ls=":", alpha=0.5)
    plt.setp(ax1.get_xticklabels(), rotation=15, ha="right", fontsize=9)

    bars = ax2.bar(by_type.index, by_type["进件转化率"] * 100,
                   color=["#F58518", "#54A24B"], width=0.5)
    for b, v in zip(bars, by_type["进件转化率"] * 100):
        ax2.text(b.get_x() + b.get_width() / 2, v + 0.5, "{:.1f}%".format(v),
                 ha="center", fontsize=10)
    ax2.set_ylim(0, max(by_type["进件转化率"] * 100) * 1.25)
    ax2.set_title("按「新老客」维度下钻")
    ax2.grid(axis="y", ls=":", alpha=0.5)
    fig.suptitle("Step2  明确分析维度：同一个指标换个维度看", fontsize=13)
    fig.tight_layout()
    save(fig, "71c_维度下钻.png")
    return df, by_type


def main():
    chart_osm()

    dic = pd.DataFrame(DICT_ROWS, columns=DICT_COLS)
    print("=" * 78)
    print("Step2 指标字典（照抄模板 Step2，共 {} 条）".format(len(dic)))
    print(dic[["层级", "指标名称", "计算规则", "值", "指标类型", "关联指标"]].to_string(index=False))

    result, order = compute_metrics(BASE_VALUES)
    print("\n[拓扑排序] 计算顺序:", " → ".join(order[:8]), "...")
    print("\n[指标结果] 基础指标 = 可调参数；派生指标 = 自动算")
    for name, (deps, _, unit) in DERIVED.items():
        v = result[name]
        txt = "{:.2f}%".format(v * 100) if unit == "%" else "{:,.0f} {}".format(v, unit)
        print("  {:<20} = {:<12}  (依赖: {})".format(name, txt, ", ".join(deps)))

    chart_dependency(result)
    df_ch, df_type = chart_dimension()
    print("\n[核对] 渠道明细合计: 注册 {} / 进件 {}".format(
        df_ch["注册用户数"].sum(), df_ch["进件用户数"].sum()))
    print("[核对] 整体进件转化率 = {:.2f}%".format(
        df_ch["进件用户数"].sum() / df_ch["注册用户数"].sum() * 100))
    print("[核对] 机器+人工审核通过 = {} + {} = {}，等于审核通过用户数 {}".format(
        BASE_VALUES["系统自动审核通过用户数"], BASE_VALUES["人工审核通过用户数"],
        BASE_VALUES["系统自动审核通过用户数"] + BASE_VALUES["人工审核通过用户数"],
        BASE_VALUES["审核通过用户数"]))
    print("\n7.1 完成：OSM 拆解 / 指标依赖计算 / 维度下钻，共 3 张图，位于:", CHART_DIR)


if __name__ == "__main__":
    main()
