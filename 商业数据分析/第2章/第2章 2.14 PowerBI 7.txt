# -*- coding: utf-8 -*-
"""
第2章 数据分析方法 —— 2.20 PowerBI：创建数据模型示例 复现
========================================================
原教材实现：PowerBI（5、PowerBI基本操作/创建数据模型示例.xlsx）
本项目迁移：Python (pandas)

【项目内涵理解】
这份文件就是“星型模型”的教科书示例：事实表 + 三张维度表，靠外键关联。
  事实表：销售业绩表（客户ID / 城市编号 / 产品编号 / 金额($M)）
  维度表：客户信息表、城市信息表、产品信息表
PowerBI 里“创建数据模型”= 把这些表用关系连起来。Python 里就是 merge。
复现：把事实表依次关联三张维度表，再按 城市 / 产品 / 客户 做销售额聚合。
"""
import os
import pandas as pd
import matplotlib.pyplot as plt

# ---------- 0. 路径配置 ----------
CASE = r"C:/Users/LENOVO/Desktop/CDA 教材资料2025/1《CDA 一级认证教材：商业数据分析》/第2章 数据分析方法/5、PowerBI基本操作"
SAVE_CHART = True
try:
    BASE_DIR = os.path.dirname(os.path.abspath(__file__))   # 作为 .py 文件运行时
except NameError:
    BASE_DIR = os.getcwd()                                  # 在 Jupyter 里粘贴运行时
CHART_DIR = os.path.join(BASE_DIR, "charts")

# ---------- 1. 读取四张表 ----------
xl = pd.ExcelFile(f"{CASE}/创建数据模型示例.xlsx")
fact = xl.parse("销售业绩表")
cust = xl.parse("客户信息表")
city = xl.parse("城市信息表")
prod = xl.parse("产品信息表")
print("各表字段：")
for name, d in [("销售业绩表", fact), ("客户信息表", cust), ("城市信息表", city), ("产品信息表", prod)]:
    print(f"  {name}: {list(d.columns)}  ({len(d)} 行)")

# ---------- 2. 星型模型：事实表关联维度表 ----------
# 维度表除主键外取第一个文本列作为“名称/类别”展示列
def name_col(d, key):
    cols = [c for c in d.columns if c != key]
    return cols[0] if cols else None

fact = fact.merge(cust, on="客户ID", how="left")
fact = fact.merge(city, on="城市编号", how="left")
fact = fact.merge(prod, on="产品编号", how="left")
print(f"\n关联后事实表行数 = {len(fact)}，字段 = {list(fact.columns)}")

plt.rcParams["font.sans-serif"] = ["SimHei", "Microsoft YaHei"]
plt.rcParams["axes.unicode_minus"] = False

# ---------- 3. 按城市聚合 ----------
print(f"\n总销售额(金额$M) = {fact['金额（$M）'].sum():.3f}")
g_city = fact.groupby(city.columns[1])["金额（$M）"].sum().sort_values(ascending=False) \
    if len(city.columns) > 1 else fact.groupby("城市编号")["金额（$M）"].sum().sort_values(ascending=False)
print("【按城市销售额 TOP10】")
print(g_city.head(10).to_string())
fig, ax = plt.subplots(figsize=(8, 4.5))
ax.bar(g_city.head(10).index.astype(str), g_city.head(10).values, color="#4C78A8")
ax.set_title("各城市销售额 TOP10"); ax.set_ylabel("金额($M)"); plt.xticks(rotation=45)
plt.tight_layout()
if SAVE_CHART:
    os.makedirs(CHART_DIR, exist_ok=True)
    fig.savefig(os.path.join(CHART_DIR, "220a_城市销售额.png"), dpi=150, bbox_inches="tight")
plt.show()

# ---------- 4. 按产品聚合 ----------
p_name = prod.columns[1] if len(prod.columns) > 1 else "产品编号"
g_prod = fact.groupby(p_name)["金额（$M）"].sum().sort_values(ascending=False).head(10)
print("\n【TOP10 产品销售额】")
print(g_prod.to_string())
fig, ax = plt.subplots(figsize=(8, 4.5))
ax.barh(g_prod.index.astype(str)[::-1], g_prod.values[::-1], color="#54A24B")
ax.set_title("TOP10 产品销售额"); ax.set_xlabel("金额($M)")
plt.tight_layout()
if SAVE_CHART:
    fig.savefig(os.path.join(CHART_DIR, "220b_TOP产品.png"), dpi=150, bbox_inches="tight")
plt.show()
