# 表名 -> (列名, CSV 文件名)。列名顺序照抄 1 create_table.sql，CSV 没有表头。

def read_csv_auto(path, columns, nrows=None):
    """教材 CSV 编码不统一：一部分是 gb2312/gb18030，一部分是 utf-8。
    逐个试，能读通就用哪个，返回 (DataFrame, 编码)。"""
    # 顺序不能反：GBK 字节用 utf-8 解会报错，可以安全退回 gb18030；
    # 但 UTF-8 字节用 gb18030 解「不会报错、只会变乱码」，所以必须先试 utf-8。
    last_err = None
    for enc in ("utf-8-sig", "gb18030", "gbk", "latin-1"):
        try:
            df = pd.read_csv(path, header=None, names=columns, encoding=enc,
                             nrows=nrows, skipinitialspace=True)
            return df, enc
        except UnicodeDecodeError as e:
            last_err = e
    raise RuntimeError("所有编码都读不了 {}: {}".format(path, last_err))


def create_and_load(conn):
    """方式1：建表 + 批量灌数据（对应 CREATE TABLE + LOAD DATA INFILE）。"""
    print("=" * 78)
    print("方式1  建表 + 批量导入")
    print("=" * 78)
    for table, (cols, fname) in TABLES.items():
        path = os.path.join(CSV_DIR, fname)
        if not os.path.exists(path):
            print("  [跳过] 找不到", fname)
            continue
        # MySQL 里每个字段都写了类型；sqlite 这里让 pandas 自己推断，够用
        conn.execute("CREATE TABLE {} ({})".format(
            table, ", ".join('"{}"'.format(c) for c in cols)))
        nrows = TRANS_NROWS if table == "Kdd99_trans" else None
        df, enc = read_csv_auto(path, cols, nrows=nrows)
        df.to_sql(table, conn, if_exists="append", index=False)
        cnt = conn.execute("SELECT COUNT(*) FROM {}".format(table)).fetchone()[0]
        print("  {:<16} {:>9,} 行  {:>2} 列  enc={}".format(table, cnt, len(cols), enc))
    return conn


def demo_insert_mode(conn):
    """方式2：教材说的 insert 脚本，Python 里就是 executemany 逐行插，慢。"""
    print("\n" + "=" * 78)
    print("方式2  逐行 INSERT（教材说全量要 1 小时，这里拿小表 One 演示）")
    print("=" * 78)
    conn.execute("CREATE TABLE one_by_insert (x INTEGER, a TEXT)")
    df, _ = read_csv_auto(os.path.join(CSV_DIR, "One.csv"), ["x", "a"])
    rows = df.where(pd.notna(df), None).values.tolist()
    conn.executemany("INSERT INTO one_by_insert VALUES (?, ?)", rows)
    got = conn.execute("SELECT COUNT(*) FROM one_by_insert").fetchone()[0]
    print("  插入 {} 行，与方式1 的 One 表一致：{}".format(
        got, got == conn.execute("SELECT COUNT(*) FROM One").fetchone()[0]))


def verify(conn):
    """跑两条教材风格的 SQL，确认库真的能用。"""
    print("\n" + "=" * 78)
    print("验证：在 sqlite 里跑教材风格的 SQL")
    print("=" * 78)

    q1 = "SELECT year, market, sale, profit FROM sale WHERE year = 2012"
    print("\n  SQL:", q1)
    print(pd.read_sql_query(q1, conn).to_string(index=False))

    q2 = ("SELECT market, SUM(sale) AS 销售额, SUM(profit) AS 利润 "
          "FROM sale GROUP BY market ORDER BY 销售额 DESC")
    print("\n  SQL:", q2)
    print(pd.read_sql_query(q2, conn).to_string(index=False))

    q3 = ("SELECT a.status, COUNT(*) AS 笔数, SUM(a.amount) AS 金额 "
          "FROM Kdd99_loans a GROUP BY a.status")
    print("\n  SQL:", q3)
    print(pd.read_sql_query(q3, conn).to_string(index=False))
