# 2) 打印能力域与能力项全景（结构化核对）
# ============================================================
def build_domain_table():
    rows = []
    for d in DCMM_DOMAINS:
        rows.append({
            "能力域": d["name"],
            "英文": d["en"],
            "能力项数量": len(d["items"]),
            "能力项": "、".join(d["items"]),
        })
    return pd.DataFrame(rows)

# 3) 图一：能力域 × 成熟度 典型达标要求矩阵（彩色热力表）
# ============================================================
def plot_domain_matrix():
    domains = [d["name"] for d in DCMM_DOMAINS]
    levels = [m["name"] for m in MATURITY_LEVELS]
    # 矩阵值：1~5，目标等级及其以下为达标(1)，以上为待提升(0)；用连续深浅表示“距离目标的远近”
    mat = np.zeros((len(domains), len(levels)))
    for i, dname in enumerate(domains):
        target = DOMAIN_TARGET[dname]
        for j, lv_obj in enumerate(MATURITY_LEVELS):
            # 已达标 = 1.0；差1级 = 0.6；差2级 = 0.3；更远 = 0.1
            diff = target - lv_obj["lv"]
            mat[i, j] = 1.0 if diff >= 0 else max(0.1, 0.6 + 0.25 * diff)
    fig, ax = plt.subplots(figsize=(9.5, 6.2))
    cmap = plt.cm.Greens
    im = ax.imshow(mat, cmap=cmap, vmin=0.05, vmax=1.0, aspect="auto")
    ax.set_xticks(range(len(levels)))
    ax.set_xticklabels(levels)
    ax.set_yticks(range(len(domains)))
    ax.set_yticklabels(domains)
    ax.set_title("DCMM 能力域 × 成熟度：典型达标要求矩阵（示意）\n"
                 "颜色越深 = 越接近/已达到该能力域的目标成熟度", fontsize=12, fontweight="bold")
    for i in range(len(domains)):
        t = DOMAIN_TARGET[domains[i]]
        for j, lv_obj in enumerate(MATURITY_LEVELS):
            reached = lv_obj["lv"] <= t
            mark = "●" if reached else "○"
            ax.text(j, i, mark, ha="center", va="center",
                    color="white" if mat[i, j] > 0.5 else "#333333", fontsize=11)
    # 右侧标注各域目标等级
    for i, dname in enumerate(domains):
        ax.text(len(levels) + 0.35, i, f"目标 L{DOMAIN_TARGET[dname]}",
                ha="left", va="center", fontsize=9, color="#1a6e2e")
    cbar = fig.colorbar(im, ax=ax, fraction=0.025, pad=0.02)
    cbar.set_label("达标程度", fontsize=9)
    ax.set_xlim(-0.5, len(levels) + 0.9)
    plt.tight_layout()
    p = os.path.join(CHART_DIR, "11.1_DCMM能力域成熟度矩阵.png")
    plt.savefig(p, dpi=130, bbox_inches="tight")
    plt.close()
    print(f"[图] 已保存：{p}")
    return p

# 4) 图二：成熟度演进路线图（阶梯爬坡）
# ============================================================
def plot_roadmap():
    fig, ax = plt.subplots(figsize=(10, 5.5))
    xs = [m["lv"] for m in MATURITY_LEVELS]
    ys = [m["lv"] for m in MATURITY_LEVELS]
    # 阶梯线
    ax.step(xs, ys, where="mid", color="#2c7fb8", linewidth=2.5)
    colors = ["#d73027", "#fc8d59", "#fee08b", "#91cf60", "#1a9850"]
    for x, y, m, c in zip(xs, ys, MATURITY_LEVELS, colors):
        ax.scatter(x, y, s=420, color=c, edgecolor="#333", zorder=3)
        ax.text(x, y + 0.18, f"L{y}\n{m['name']}", ha="center", va="bottom", fontsize=9.5, fontweight="bold")
    # 阶段说明
    phase_notes = {
        1: "被动、项目级",
        2: "有流程、专人管",
        3: "组织级标准化",
        4: "量化监控",
        5: "持续最优",
    }
    for x, y in zip(xs, ys):
        ax.text(x, y - 0.42, phase_notes[x], ha="center", va="top", fontsize=8, color="#555")
    ax.set_xlim(0.4, 5.6)
    ax.set_ylim(0.3, 5.7)
    ax.set_xticks(xs)
    ax.set_xlabel("成熟度等级（L1→L5）", fontsize=10)
    ax.set_ylabel("能力建设水平", fontsize=10)
    ax.set_title("DCMM 数据管理能力成熟度演进路线图", fontsize=13, fontweight="bold")
    ax.grid(True, axis="y", linestyle=":", alpha=0.4)
    ax.set_axisbelow(True)
    plt.tight_layout()
    p = os.path.join(CHART_DIR, "11.1_成熟度演进路线图.png")
    plt.savefig(p, dpi=130, bbox_inches="tight")
    plt.close()
    print(f"[图] 已保存：{p}")
    return p

# 5) 图三：能力域能力项数量条形图（看各域建设工作量）
# ============================================================
def plot_item_counts():
    domains = [d["name"] for d in DCMM_DOMAINS]
    counts = [len(d["items"]) for d in DCMM_DOMAINS]
    fig, ax = plt.subplots(figsize=(10, 5))
    bars = ax.barh(domains, counts, color="#4292c6", edgecolor="#1f5c8a")
    for b, c in zip(bars, counts):
        ax.text(b.get_width() + 0.05, b.get_y() + b.get_height() / 2, str(c),
                va="center", fontsize=10, fontweight="bold")
    ax.set_xlabel("能力项数量", fontsize=10)
    ax.set_title("DCMM 各能力域下属能力项数量", fontsize=13, fontweight="bold")
    ax.invert_yaxis()
    ax.set_xlim(0, max(counts) + 0.8)
    plt.tight_layout()
    p = os.path.join(CHART_DIR, "11.1_能力项数量.png")
    plt.savefig(p, dpi=130, bbox_inches="tight")
    plt.close()
    print(f"[图] 已保存：{p}")
    return p
