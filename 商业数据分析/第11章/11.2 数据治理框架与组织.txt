import os
import matplotlib.pyplot as plt
from matplotlib.patches import FancyBboxPatch, FancyArrowPatch

try:
    BASE_DIR = os.path.dirname(os.path.abspath(__file__))
except NameError:
    BASE_DIR = os.getcwd()
CHART_DIR = os.path.join(BASE_DIR, "charts")
os.makedirs(CHART_DIR, exist_ok=True)

plt.rcParams["font.sans-serif"] = ["SimHei", "Microsoft YaHei", "Arial"]
plt.rcParams["axes.unicode_minus"] = False

_BOX = dict(boxstyle="round,pad=0.5", linewidth=1.2)
BLUE = "#2c7fb8"
GREEN = "#31a354"
ORANGE = "#e6550d"
GREY = "#756bb1"


def _box(ax, x, y, w, h, text, fc, ec="#333", fs=10, tc="white"):
    ax.add_patch(FancyBboxPatch((x - w / 2, y - h / 2), w, h,
                                boxstyle="round,pad=0.4,rounding_size=0.08",
                                linewidth=1.2, edgecolor=ec, facecolor=fc))
    ax.text(x, y, text, ha="center", va="center", fontsize=fs,
            fontweight="bold", color=tc, wrap=True)


# ============================================================
# 图1：数据治理 vs 数据管理（定位与区别）
# ============================================================
def plot_gov_vs_mgmt():
    fig, ax = plt.subplots(figsize=(10, 5.2))
    ax.set_xlim(0, 10); ax.set_ylim(0, 6); ax.axis("off")
    # 两个大圈示意“治理”管方向/责任，“管理”管执行/操作，二者交叠于“数据”
    from matplotlib.patches import Circle
    c1 = Circle((3.6, 3), 2.1, color=BLUE, alpha=0.18, ec=BLUE, lw=2)
    c2 = Circle((6.4, 3), 2.1, color=GREEN, alpha=0.18, ec=GREEN, lw=2)
    ax.add_patch(c1); ax.add_patch(c2)
    ax.text(3.6, 4.9, "数据治理\n(Data Governance)", ha="center", fontsize=12, fontweight="bold", color=BLUE)
    ax.text(6.4, 4.9, "数据管理\n(Data Management)", ha="center", fontsize=12, fontweight="bold", color=GREEN)
    ax.text(3.6, 3.2, "管“方向、责任、\n政策与决策”", ha="center", fontsize=9.5, color="#1f4e79")
    ax.text(6.4, 3.2, "管“执行、操作、\n流程与工具”", ha="center", fontsize=9.5, color="#1a6e2e")
    ax.text(5.0, 2.4, "交叠：统一的数据\n标准 / 质量 / 安全", ha="center", fontsize=9, color="#555")
    notes = [
        "治理：由董事会/高管层负责，定战略、明权责、建制度（见银行指引第8-14条）",
        "管理：在日常运营中执行采集、存储、加工、共享、质量检查等操作",
        "关系：治理为管理提供框架与依据，管理是治理落地的手段",
    ]
    for i, t in enumerate(notes):
        ax.text(0.3, 0.9 - i * 0.55, "- " + t, ha="left", va="center", fontsize=9, color="#333")
    ax.set_title("数据治理 与 数据管理：定位与关系", fontsize=13, fontweight="bold")
    plt.tight_layout()
    p = os.path.join(CHART_DIR, "11.2_治理与管理关系.png")
    plt.savefig(p, dpi=130, bbox_inches="tight"); plt.close()
    print(f"[图] 已保存：{p}")
    return p


# ============================================================
# 图2：银行业金融机构数据治理架构（依据银行指引第8-14条原文）
# ============================================================
def plot_bank_governance_org():
    fig, ax = plt.subplots(figsize=(9.5, 7))
    ax.set_xlim(0, 10); ax.set_ylim(0, 11); ax.axis("off")
    layers = [
        (5, 10.0, 9.4, 1.1, "董事会", "制定数据战略；审批重大事项；对数据治理承担最终责任（第9条）", BLUE),
        (5, 8.6, 9.4, 1.1, "监事会", "对董事会、高级管理层在数据治理方面的履职情况进行监督评价（第10条）", "#6a51a3"),
        (5, 7.2, 9.4, 1.1, "高级管理层", "建立数据治理体系；配置资源；可设首席数据官(CDO)；定期向董事会报告（第11条）", GREEN),
        (5, 5.5, 9.4, 1.3, "归口管理部门", "牵头建设数据治理体系；协调运行机制；推动数据应用；设监管数据专职岗（第12、14条）", ORANGE),
        (5, 3.4, 9.4, 1.6, "业务部门", "负责本业务领域数据治理；管理业务条线数据源；落实数据质量控制（第13条）", "#3182bd"),
        (5, 1.4, 9.4, 1.1, "专业队伍 / 专职·兼职岗位", "满足数据治理需要的专业队伍；按年度培训；规划职业通道（第14、15条）", "#969696"),
    ]
    for x, y, w, h, title, desc, c in layers:
        _box(ax, x, y, w, h, "", c)
        ax.text(x, y + h / 2 - 0.28, title, ha="center", va="center",
                fontsize=12, fontweight="bold", color="white")
        ax.text(x, y - h / 2 + 0.32, desc, ha="center", va="center",
                fontsize=8.2, color="white")
    # 连接箭头（自上而下职责传导）
    for y_top in [9.45, 8.05, 6.55, 4.7, 2.55]:
        ax.add_patch(FancyArrowPatch((5, y_top), (5, y_top - 0.35),
                      arrowstyle="-|>", mutation_scale=14, color="#444", lw=1.4))
    ax.set_title("银行业金融机构数据治理架构（据银保监发〔2018〕22号 第8-15条）",
                 fontsize=12.5, fontweight="bold")
    plt.tight_layout()
    p = os.path.join(CHART_DIR, "11.2_银行治理架构.png")
    plt.savefig(p, dpi=130, bbox_inches="tight"); plt.close()
    print(f"[图] 已保存：{p}")
    return p


# ============================================================
# 图3：数据治理内容框架（依据银行指引各章 + DCMM 能力域）
# ============================================================
def plot_content_framework():
    fig, ax = plt.subplots(figsize=(10, 7))
    ax.set_xlim(0, 10); ax.set_ylim(0, 10); ax.axis("off")
    # 顶层：数据战略
    _box(ax, 5, 9.0, 8.6, 1.1, "数据战略（治理目标与方向，银行指引第17条）", BLUE)
    # 中层：6 大治理内容域（第一行 4 个、第二行 2 个，互不重叠）
    domains = [
        ("数据制度", "组织/职责/协调/监督\n（第18-19条）", "#3182bd"),
        ("数据标准", "统一业务规范与技术标准\n（第20条）", "#31a354"),
        ("数据质量", "真实/准确/完整/及时\n监控考核（第29-37条）", "#e6550d"),
        ("数据安全", "分级/权限/隐私/审计\n（第24条）", "#c51b8a"),
        ("数据生命周期", "采集共享/系统/资料/应急\n（第21-26条）", "#756bb1"),
        ("数据应用", "风险/经营/内控\n的数据驱动（第38-49条）", "#0868ac"),
    ]
    positions = [(1.35, 6.7), (3.75, 6.7), (6.15, 6.7), (8.55, 6.7), (2.85, 4.35), (6.85, 4.35)]
    for (px, py), (t, d, c) in zip(positions, domains):
        _box(ax, px, py, 2.2, 1.6, "", c)
        ax.text(px, py + 0.45, t, ha="center", va="center", fontsize=10, fontweight="bold", color="white")
        ax.text(px, py - 0.25, d, ha="center", va="center", fontsize=7.2, color="white")
    # 底层：数据价值实现 + 自我评估/问责机制
    _box(ax, 2.85, 2.0, 3.6, 1.3, "数据价值实现\n（数据驱动经营，第38-49条）", GREEN)
    _box(ax, 6.85, 2.0, 3.6, 1.3, "自我评估 + 问责/激励\n（第27-28条，年度报送监管）", ORANGE)
    # 箭头
    ax.add_patch(FancyArrowPatch((5, 8.45), (5, 7.6), arrowstyle="-|>", mutation_scale=14, color="#444"))
    ax.add_patch(FancyArrowPatch((4.9, 3.5), (3.3, 2.7), arrowstyle="-|>", mutation_scale=12, color="#444"))
    ax.add_patch(FancyArrowPatch((5.1, 3.5), (6.4, 2.7), arrowstyle="-|>", mutation_scale=12, color="#444"))
    ax.set_title("数据治理内容框架（银行业金融机构，据指引第2-5章）", fontsize=12.5, fontweight="bold")
    plt.tight_layout()
    p = os.path.join(CHART_DIR, "11.2_治理内容框架.png")
    plt.savefig(p, dpi=130, bbox_inches="tight"); plt.close()
    print(f"[图] 已保存：{p}")
    return p

# 图4：IBM 数据治理统一流程 14 步（10 必需 + 4 可选专题）
# ============================================================
def plot_ibm_process():
    # 10 个必需步骤（据 PDF 目录第3-11章 + 文末最后必需步骤“监视和度量”）
    required = [
        "1 定义业务问题", "2 获取高层支持", "3 执行成熟度评估",
        "4 创建路线图", "5 建立组织蓝图", "6 创建数据字典",
        "7 理解数据", "8 创建元数据存储库", "9 定义度量指标",
        "10 监视和度量",
    ]
    # 4 个可选专题（PDF 第405-406、519、537、575行）
    optional = [
        "★ 主数据治理（实现主数据管理）",
        "★ 分析治理（指导分析）",
        "★ 安全和隐私",
        "★ 信息生命周期治理",
    ]
    fig, ax = plt.subplots(figsize=(11, 6.8))
    ax.set_xlim(0, 12); ax.set_ylim(0, 11); ax.axis("off")
    ax.text(6, 10.4, "IBM 数据治理统一流程（14 步 = 10 必需 + 4 可选专题）",
            ha="center", fontsize=13, fontweight="bold")
    # 必需步骤两列
    ax.text(3, 9.5, "10 个必需步骤（奠定治理基础，不可或缺）", ha="center", fontsize=10.5, color=BLUE, fontweight="bold")
    for i, s in enumerate(required):
        col = i // 5
        row = i % 5
        x = 1.7 + col * 3.6
        y = 8.4 - row * 1.45
        _box(ax, x, y, 3.1, 1.1, s, BLUE, fs=9.5)
    # 可选专题
    ax.text(9.1, 9.5, "4 个可选专题（按需要选 1+）", ha="center", fontsize=10.5, color=ORANGE, fontweight="bold")
    for i, s in enumerate(optional):
        y = 8.4 - i * 1.6
        _box(ax, 9.1, y, 3.4, 1.25, s, ORANGE, fs=8.6)
    # 底部循环说明（PDF 原文：整个流程需以持续循环的形式操作，度量结果用于改进）
    ax.add_patch(FancyArrowPatch((1.7, 0.75), (9.1, 0.75),
                  arrowstyle="-|>", mutation_scale=16, color="#444", lw=1.6))
    ax.text(5.4, 0.22, "度量结果回流至第 1 步 —— 整个流程以持续循环方式运行（度量 → 改进）",
            ha="center", fontsize=9, color="#444")
    ax.set_title("", fontsize=1)
    plt.tight_layout()
    p = os.path.join(CHART_DIR, "11.2_IBM统一流程.png")
    plt.savefig(p, dpi=130, bbox_inches="tight"); plt.close()
    print(f"[图] 已保存：{p}")
    return p

    plot_gov_vs_mgmt()

    print("\n11.2 完成：数据治理框架与组织复现完毕。")
