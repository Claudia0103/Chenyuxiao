def _card(ax, x, y, w, h, title, color, lines):
    ax.add_patch(FancyBboxPatch((x, y), w, h,
                 boxstyle="round,pad=0.02,rounding_size=0.1",
                 fc="#FFFFFF", ec=color, lw=2))
    ax.add_patch(FancyBboxPatch((x, y + h - 0.6), w, 0.6,
                 boxstyle="round,pad=0.02,rounding_size=0.1",
                 fc=color, ec="none"))
    ax.text(x + w / 2, y + h - 0.3, title, ha="center", va="center",
            fontsize=12.5, color="white", fontweight="normal")
    ty = y + h - 1.05
    for ln in lines:
        ax.text(x + 0.25, ty, ln, ha="left", va="center", fontsize=10, color="#222222")
        ty -= 0.5

def _layout(node, depth, leaf_counter, pos):
    if "children" not in node or not node["children"]:
        y = leaf_counter[0] * 0.62
        leaf_counter[0] += 1
        pos[id(node)] = (depth, y)
        return y
    ys = []
    for c in node["children"]:
        ys.append(_layout(c, depth + 1, leaf_counter, pos))
    y = sum(ys) / len(ys)
    pos[id(node)] = (depth, y)
    return y


def plot_dimension_tree():
    pos = {}
    _layout(DIM_TREE, 0, [0], pos)
    max_depth = max(d for d, _ in pos.values())
    n_leaves = max(y for _, y in pos.values()) / 0.62 + 1
    dx = 2.7

    fig, ax = plt.subplots(figsize=(15, max(7, n_leaves * 0.62 + 1)))
    ax.set_xlim(-0.3, max_depth * dx + 3.6)
    ax.set_ylim(-0.5, n_leaves * 0.62 + 0.5)
    ax.axis("off")

    # 连线
    for node in _iter(DIM_TREE):
        if "children" in node and node["children"]:
            xd, yd = pos[id(node)]
            for c in node["children"]:
                xc, yc = pos[id(c)]
                ax.plot([xd + 0.9, xc - 0.05], [yd, yc], color="#9AA5B1", lw=1.2, zorder=1)
    # 节点
    for node in _iter(DIM_TREE):
        d, y = pos[id(node)]
        x = d * dx
        is_leaf = "children" not in node or not node["children"]
        color = C_BLUE if d == 0 else (C_TEAL if is_leaf else C_GREEN)
        w = 2.5 if not is_leaf else 3.1
        ax.add_patch(FancyBboxPatch((x, y - 0.22), w, 0.44,
                     boxstyle="round,pad=0.02,rounding_size=0.06",
                     fc=color, ec="none", zorder=2))
        ax.text(x + w / 2, y, node["name"], ha="center", va="center",
                fontsize=10 if not is_leaf else 9.5,
                color="white", zorder=3,
                linespacing=1.2)

    ax.set_title("账户维度树（维度树-账户维度）：账户 → 存款账户 → 个人存款账户 → … → 具体账户",
                 fontweight="normal", pad=12)
    p = os.path.join(OUT, "13.3_账户维度树.png")
    fig.savefig(p, dpi=130, bbox_inches="tight")
    plt.close(fig)
    return p


def _iter(node):
    yield node
    for c in node.get("children", []):
        yield from _iter(c)

# 6. 可运行：指标标准 + 维度树
# ======================================================================
class IndicatorStandard:
    REQUIRED = ["编号", "中文名称", "指标类型", "技术口径"]

    def __init__(self, **kw):
        self.f = kw

    def is_derived(self):
        return self.f.get("指标类型") == "派生指标"

    def validate(self):
        miss = [k for k in self.REQUIRED if not self.f.get(k)]
        if self.is_derived() and not self.f.get("关联标准"):
            miss.append("关联标准(派生指标必填)")
        return miss


class DimensionTree:
    def __init__(self, node):
        self.root = node

    def depth(self):
        def d(n):
            return 1 if not n.get("children") else 1 + max(d(c) for c in n["children"])
        return d(self.root)

    def leaves(self):
        out = []
        def walk(n):
            if not n.get("children"):
                out.append(n["name"])
            else:
                for c in n["children"]:
                    walk(c)
        walk(self.root)
        return out

# 7. 主程序
# ======================================================================
def main():
    g = IndicatorStandard(**IND_G)
    p = IndicatorStandard(**IND_P)
    print("DZ-G-014 校验:", g.validate(), "| 是否派生:", g.is_derived())
    print("DZ-P-121 校验:", p.validate(), "| 是否派生:", p.is_derived())

    tree = DimensionTree(DIM_TREE)
    print(f"账户维度树：深度={tree.depth()}, 叶子(具体账户)={len(tree.leaves())} 个")
    for name in tree.leaves():
        print("   -", name)

    paths = [plot_indicator_attr_model(), plot_root_derived(),
             plot_dimension_tree(), plot_dimension_overview()]
    print("\n图表已生成：")
    for pp in paths:
        print("  -", pp)
    return paths


if __name__ == "__main__" or True:
    main()
