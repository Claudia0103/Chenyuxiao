# 2. 绘图辅助
# ======================================================================
def _box(ax, x, y, w, h, text, fc, tc="white", fs=11, lh=1.35, weight="normal"):
    ax.add_patch(FancyBboxPatch((x, y), w, h,
                 boxstyle="round,pad=0.02,rounding_size=0.08",
                 fc=fc, ec="none", zorder=2))
    ax.text(x + w / 2, y + h / 2, text, ha="center", va="center",
            fontsize=fs, color=tc, zorder=3, linespacing=lh, fontweight=weight)


def _arrow(ax, x1, y1, x2, y2, color=C_GRAY, style="-|>"):
    ax.annotate("", xy=(x2, y2), xytext=(x1, y1),
                arrowprops=dict(arrowstyle=style, color=color, lw=1.6),
                zorder=1)

# 3. 图1：数据标准管理 5 阶段生命周期
# ======================================================================
def plot_lifecycle():
    fig, ax = plt.subplots(figsize=(15, 5.2))
    ax.set_xlim(0, 15)
    ax.set_ylim(0, 5.4)
    ax.axis("off")

    n = len(STAGES)
    slot = 15.0 / n
    bw, bh = slot * 0.78, 1.5
    cy = 3.6
    centers = []
    for i, (name, color, subs) in enumerate(STAGES):
        cx = slot * i + slot / 2
        x0 = cx - bw / 2
        _box(ax, x0, cy - bh / 2, bw, bh, name, color, fs=13, weight="normal")
        centers.append(cx)
        # 子活动（3 行）
        sub_txt = "\n".join("· " + s for s in subs)
        ax.text(cx, cy - bh / 2 - 1.05, sub_txt, ha="center", va="top",
                fontsize=10, color="#333333", linespacing=1.5, zorder=3)
        # 阶段编号
        ax.text(x0 + 0.18, cy + bh / 2 - 0.2, f"{i+1}", ha="left", va="top",
                fontsize=12, color="white", fontweight="normal", zorder=4)

    # 阶段间箭头
    for i in range(n - 1):
        x1 = centers[i] + bw / 2
        x2 = centers[i + 1] - bw / 2
        _arrow(ax, x1 + 0.05, cy, x2 - 0.05, cy, color=C_GRAY)

    ax.set_title("数据标准管理生命周期：5 个阶段（规划→建设→评审发布→落地执行→运营维护）",
                 fontweight="normal", pad=12)
    p = os.path.join(OUT, "13.1_管理生命周期五阶段.png")
    fig.savefig(p, dpi=130, bbox_inches="tight")
    plt.close(fig)
    return p

# 4. 图2：3 层组织架构
# ======================================================================
def plot_org():
    fig, ax = plt.subplots(figsize=(13, 5.6))
    ax.set_xlim(0, 13)
    ax.set_ylim(0, 6)
    ax.axis("off")

    bw, bh = 11.0, 1.35
    x0 = 1.0
    ys = [4.4, 2.7, 1.0]  # 决策 / 管理 / 执行
    for (name, color, desc), y in zip(ORG, ys):
        _box(ax, x0, y, bw, bh, "", color, tc=color)
        # 标题（框内左侧）
        ax.text(x0 + 0.25, y + bh - 0.32, name, ha="left", va="center",
                fontsize=13, color="white", fontweight="normal")
        # 职责（框内，自动换行）
        ax.text(x0 + 0.25, y + bh - 0.62, _wrap(desc, 46), ha="left", va="top",
                fontsize=9.5, color="white", linespacing=1.35)
        if y < 4.4:
            _arrow(ax, x0 + bw / 2, y + bh, x0 + bw / 2, y + bh + (4.4 - 2.7 - bh),
                   color=C_GRAY)

    ax.set_title("数据标准管理组织架构：决策层 → 管理层 → 执行/专家层",
                 fontweight="normal", pad=12)
    p = os.path.join(OUT, "13.1_三层组织架构.png")
    fig.savefig(p, dpi=130, bbox_inches="tight")
    plt.close(fig)
    return p

def _wrap(text, width):
    """按字符宽度简单折行（中文按字计）。"""
    out, line, cnt = [], "", 0
    for ch in text:
        line += ch
        cnt += 1
        if cnt >= width and ch not in "，、；。（）()":
            out.append(line)
            line, cnt = "", 0
    if line:
        out.append(line)
    return "\n".join(out)


# ======================================================================
# 7. 可运行：数据标准治理阶段模型
# ======================================================================
class DataStandardGovernance:
    """用代码固化白皮书的管理框架，可打印阶段清单与流转顺序。"""

    def __init__(self):
        self.stages = [s[0] for s in STAGES]
        self.detail = {s[0]: s[2] for s in STAGES}

    def report(self):
        print("=" * 56)
        print("数据标准管理 —— 5 阶段生命周期")
        print("=" * 56)
        for i, s in enumerate(self.stages, 1):
            print(f"{i}. {s}")
            for a in self.detail[s]:
                print(f"     - {a}")
        print("-" * 56)
        print("流转：", " → ".join(self.stages))
        print("=" * 56)
        return self.stages


# ======================================================================
# 8. 主程序
# ======================================================================
def main():
    paths = [
        plot_lifecycle(),
        plot_org(),
        plot_files(),
        plot_classification(),
    ]
    gov = DataStandardGovernance()
    gov.report()
    print("\n图表已生成：")
    for p in paths:
        print("  -", p)
    return paths


if __name__ == "__main__" or True:
    main()
