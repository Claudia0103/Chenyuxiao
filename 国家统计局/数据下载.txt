# -*- coding: utf-8 -*-
"""
可视化大屏·宏观经济 —— B1 下载数据
====================================
数据源：世界银行公开数据 API（World Bank Open Data）
  https://api.worldbank.org/v2/country/<国家>/indicator/<指标>?format=json
无需注册、无需登录、无需 API Key，返回 JSON 可直接下载；也可把 format 改成 csv
用浏览器直接下载 CSV。数据由世界银行按年发布，本脚本抓取的是 1960-2025 全序列。

为什么用它：国家统计局新站（data.stats.gov.cn）的 WAF 对批量请求会弹 JS 挑战，
不适合作为「随手能下载」的演示数据源。世界银行这套接口稳定、口径统一、可复现。

本脚本做两件事：
  1) 依次请求 12 个宏观指标 × 5 个国家（每个指标一次请求，共 12 次）；
  2) 输出 data/世界银行_宏观指标_长表.csv 与 ..._宽表.csv（宽表方便 Excel 看）。
零第三方依赖，只用标准库；把整份代码粘进 Jupyter 一个 cell 就能跑。
想换国家/指标/年份，只改下面三个常量。
"""

import os
import csv
import json
import time
import urllib.parse
import urllib.request

# ---- 可调参数 ----
COUNTRIES = {
    "CHN": "中国",
    "USA": "美国",
    "JPN": "日本",
    "IND": "印度",
    "DEU": "德国",
}
INDICATORS = {
    "NY.GDP.MKTP.CD":      ("GDP总量(现价美元)", "美元"),
    "NY.GDP.MKTP.KD.ZG":   ("GDP实际增长率", "%"),
    "NY.GDP.PCAP.CD":      ("人均GDP(现价美元)", "美元"),
    "SP.POP.TOTL":         ("总人口", "人"),
    "SP.URB.TOTL.IN.ZS":   ("城镇化率", "%"),
    "SP.DYN.LE00.IN":      ("出生时预期寿命", "岁"),
    "FP.CPI.TOTL.ZG":      ("CPI通胀率", "%"),
    "NV.AGR.TOTL.ZS":      ("农业增加值占GDP", "%"),
    "NV.IND.TOTL.ZS":      ("工业增加值占GDP", "%"),
    "NV.SRV.TOTL.ZS":      ("服务业增加值占GDP", "%"),
    "NE.EXP.GNFS.ZS":      ("出口占GDP", "%"),
    "NE.IMP.GNFS.ZS":      ("进口占GDP", "%"),
}
START_YEAR = 1960
END_YEAR = 2025
SLEEP = 0.4          # 每次请求间隔，礼貌抓取
RETRIES = 3          # 单指标失败重试次数
API = "https://api.worldbank.org/v2/country/{}/indicator/{}"
UA = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 " \
     "(KHTML, like Gecko) Chrome/157.0.0.0 Safari/537.36"

try:
    _HERE = os.path.dirname(os.path.abspath(__file__))
except NameError:
    _HERE = os.getcwd()
OUT = os.path.join(_HERE, "data")
os.makedirs(OUT, exist_ok=True)


def fetch_indicator(code, retries=RETRIES):
    """请求一个指标的多国多年数据，返回原始记录列表；失败返回空列表。"""
    url = API.format(";".join(COUNTRIES), code) + "?" + urllib.parse.urlencode({
        "format": "json", "per_page": 20000,
        "date": f"{START_YEAR}:{END_YEAR}",
    })
    for k in range(retries):
        try:
            req = urllib.request.Request(
                url, headers={"User-Agent": UA, "Accept": "application/json"})
            with urllib.request.urlopen(req, timeout=40) as resp:
                payload = json.loads(resp.read().decode("utf-8"))
            if not isinstance(payload, list) or len(payload) < 2 or not payload[1]:
                raise ValueError("返回为空")
            return payload[1]
        except Exception as e:
            wait = 2.0 * (k + 1)
            print(f"  [{code}] 第 {k+1}/{retries} 次失败：{e}，{wait}s 后重试")
            time.sleep(wait)
    print(f"  [{code}] 放弃，该指标无数据")
    return []


def to_rows(items, code, cn_name, unit):
    """把 API 记录整理成长表行。缺失值(None)写成空字符串。"""
    rows = []
    for it in items:
        iso = it.get("countryiso3code") or ""
        year = str(it.get("date") or "")
        value = it.get("value")
        rows.append({
            "国家代码": iso,
            "国家": COUNTRIES.get(iso, (it.get("country") or {}).get("value", iso)),
            "指标代码": code,
            "指标": cn_name,
            "年份": int(year) if year.isdigit() else year,
            "数值": "" if value is None else value,
            "单位": unit,
        })
    return rows


def save_csv(path, fieldnames, rows):
    with open(path, "w", newline="", encoding="utf-8-sig") as f:
        w = csv.DictWriter(f, fieldnames=fieldnames, extrasaction="ignore")
        w.writeheader()
        w.writerows(rows)


def pivot_to_wide(rows):
    """长表转宽表：一行 = 一个国家 + 一年，列 = 各指标中文名。"""
    fields = ["国家代码", "国家", "年份"] + [v[0] for v in INDICATORS.values()]
    buf = {}
    for r in rows:
        key = (r["国家代码"], r["年份"])
        if key not in buf:
            buf[key] = {"国家代码": r["国家代码"], "国家": r["国家"], "年份": r["年份"]}
        buf[key][r["指标"]] = r["数值"]
    return fields, [buf[k] for k in sorted(buf, key=lambda x: (x[0], str(x[1])))]


def main():
    rows = []
    print(f"下载 {len(INDICATORS)} 个指标 × {len(COUNTRIES)} 国 "
          f"× {START_YEAR}-{END_YEAR} ...\n")
    for i, (code, (cn, unit)) in enumerate(INDICATORS.items(), 1):
        items = fetch_indicator(code)
        got = to_rows(items, code, cn, unit)
        rows.extend(got)
        print(f"  [{i:2d}/{len(INDICATORS)}] {cn:<22s} {len(got):4d} 条")
        time.sleep(SLEEP)

    long_path = os.path.join(OUT, "世界银行_宏观指标_长表.csv")
    wide_path = os.path.join(OUT, "世界银行_宏观指标_宽表.csv")
    save_csv(long_path, ["国家代码", "国家", "指标代码", "指标", "年份", "数值", "单位"], rows)
    fields, wide = pivot_to_wide(rows)
    save_csv(wide_path, fields, wide)

    print(f"\n长表 {len(rows)} 行 -> {long_path}")
    print(f"宽表 {len(wide)} 行 -> {wide_path}")

    # 打印最新年中国的几个关键数字，便于直接核对
    latest = {}
    for r in rows:
        if r["国家代码"] == "CHN" and r["数值"] != "":
            y = int(r["年份"])
            if r["指标"] not in latest or y > latest[r["指标"]][0]:
                latest[r["指标"]] = (y, r["数值"])
    print("\n中国各指标最新值：")
    for cn, _ in INDICATORS.values():
        if cn in latest:
            y, v = latest[cn]
            print(f"  {cn:<22s} {y}  {v}")
    return rows


if __name__ == "__main__" or True:
    main()
