"""
可视化大屏·宏观经济 —— B2 生成可视化大屏
==========================================
读取 B1 下载的「世界银行_宏观指标_长表.csv」，生成一个单文件 HTML：
  - ECharts 与全部数据都内联在 HTML 里，不依赖 CDN，断网也能双击打开；
  - 6 张 KPI 卡 + 9 张图表，深色科技风，顶部有年份滑块（可自动播放）和明暗切换；
  - 年份滑块同时驱动 KPI 卡与「五国 GDP 排行」，其余图表展示完整历史序列。

零第三方依赖（标准库 csv/json/os/urllib）。整份代码粘进 Jupyter 一个 cell 也能跑。
运行前先跑 B1；若 data/echarts.min.js 不存在，脚本会自动从 unpkg 下载一次。
"""

COUNTRIES = {
    "CHN": "中国", "USA": "美国", "JPN": "日本", "IND": "印度", "DEU": "德国",
}

ECHARTS_URL = "https://unpkg.com/echarts@5.5.1/dist/echarts.min.js"

try:
    _HERE = os.path.dirname(os.path.abspath(__file__))
except NameError:
    _HERE = os.getcwd()
DATA_DIR = os.path.join(_HERE, "data")
os.makedirs(DATA_DIR, exist_ok=True)


def load_long(path):
    with open(path, "r", encoding="utf-8-sig", newline="") as f:
        return list(csv.DictReader(f))


def build_payload(rows):
    """长表 -> 前端用的 {years, countries, series[国家][指标][年份]=值, latestYear}"""
    series, years = {}, set()
    for r in rows:
        y, v = str(r["年份"]), r["数值"]
        if not y.isdigit() or v == "":
            continue
        year = int(y)
        years.add(year)
        iso, ind = r["国家代码"], r["指标"]
        series.setdefault(iso, {}).setdefault(ind, {})[year] = round(float(v), 4)
    ys = sorted(years)
    # 最新年：中国 GDP 有值的最后一年，避免用某个指标缺失的年份
    latest = max(y for y in ys if "CHN" in series
                 and series["CHN"].get("GDP总量(现价美元)", {}).get(y) is not None)
    return {
        "years": ys,
        "latestYear": latest,
        "countries": [{"code": k, "name": v} for k, v in COUNTRIES.items()],
        "series": series,
        "source": "World Bank Open Data | api.worldbank.org",
        "generated": datetime.datetime.now().strftime("%Y-%m-%d %H:%M"),
    }


def get_echarts():
    path = os.path.join(DATA_DIR, ECHARTS_JS)
    if not os.path.exists(path) or os.path.getsize(path) < 300000:
        print("本地未找到 ECharts，正在下载一次 ...")
        req = urllib.request.Request(ECHARTS_URL, headers={"User-Agent": "Mozilla/5.0"})
        with urllib.request.urlopen(req, timeout=60) as r:
            blob = r.read()
        with open(path, "wb") as f:
            f.write(blob)
    with open(path, "r", encoding="utf-8", errors="replace") as f:
        text = f.read()
    return text.replace("</script", "<\\/script")

function P(){ return state.light

function val(iso, ind, y){
  const s = DATA.series[iso] && DATA.series[iso][ind];
  return (s && s[y] !== undefined) ? s[y] : null;
}
function arr(iso, ind){ return DATA.years.map(y => val(iso, ind, y)); }
function yearsFrom(a){ return DATA.years.filter(y => y >= a); }

function axis(extra){
  const p = P();

function base(extra){

function gradient(c1, c2){
  return new echarts.graphic.LinearGradient(0,0,0,1,[{offset:0,color:c1},{offset:1,color:c2}]);
}
function lineSeries(name, data, color, opts){
  return Object.assign({name:name, type:'line', data:data, smooth:.25, showSymbol:false,
    symbol:'circle', symbolSize:5, connectNulls:false,
    lineStyle:{width:2, color:color}, itemStyle:{color:color}}, opts || {});
}

const KPIS = [
  {t:'GDP 总量', ind:'GDP总量(现价美元)', unit:'万亿美元', s:1e12, d:2, kind:'pct'},
  {t:'GDP 实际增长', ind:'GDP实际增长率', unit:'%', s:1, d:2, kind:'pp'},

  {t:'总人口', ind:'总人口', unit:'亿人', s:1e8, d:2, kind:'pct'},

];
function latestAvailable(iso, ind, y){
  for (let yy = y; yy >= DATA.years[0]; yy--){
    const v = val(iso, ind, yy);
    if (v !== null) return {year:yy, value:v};
  }
  return null;
}
function renderKpis(){
  const y = state.year;
  $('kpis').innerHTML = KPIS.map(k => {
    // 个别指标（如预期寿命）最新年份尚未发布，回退到最近有数据的年份并标注
    const hit = latestAvailable(CN, k.ind, y);
    const cur = hit ? hit.value : null;
    const used = hit ? hit.year : y;
    const old = hit ? val(CN, k.ind, used - 1) : null;
    const main = cur === null ? '—' : (cur / k.s).toLocaleString('zh-CN',
      {minimumFractionDigits:k.d, maximumFractionDigits:k.d});
    let delta = '较上年 —';
    if (cur !== null && old !== null && old !== 0){
      const d = k.kind === 'pp' ? (cur - old) : (cur/old - 1)*100;
      const unit = k.kind === 'pp' ? ' pp' : '%';
      const cls = d >= 0 ? 'up' : 'down';
      delta = '较上年 <span class="'+cls+'">'+(d>=0?'+':'')+d.toFixed(2)+unit+'</span>';
    }
    return '<div class="kpi"><div class="t">'+k.t+' · '+used+'</div>'+
           '<div class="v">'+main+'<small>'+k.unit+'</small></div>'+
           '<div class="d">'+delta+'</div></div>';
  }).join('');
}
