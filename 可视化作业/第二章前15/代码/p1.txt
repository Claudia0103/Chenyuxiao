DPI = 120.0                
EM_PT = DPI / 72.0         
SIZE_PX = [640, 507]       
PLOT_PX = [0.144, 0.3189, 0.7602, 0.5254]  
BG = "#1A1E43"           
BLUE = "#0070C0"           
CYAN = "#00B0F0"            
TXT = "#FFFFFF"             
NOTE = "#D9D9D9"           
GRID = "#404361"            
GREY = "#9AAEC8"
CATS = ["华北", "华南", "东北", "西北", "西南", "华东"]        
VALS = [2354.0, 1902.0, 3524.0, 2698.0, 2896.0, 2563.0]      
_FONT_FILES = {

    "ea": {"n": "C:/Windows/Fonts/msyh.ttc", "b": "C:/Windows/Fonts/msyhbd.ttc"},

    for _p in _d.values():

_CJK_RE = re.compile(r"[\u3000-\u303F\u4E00-\u9FFF\uFF00-\uFF60]")
_FP_CACHE = {}
def fp(sz, bold=False, ea=False):
    key = (round(float(sz), 2), bool(bold), bool(ea))
    p = _FP_CACHE.get(key)
    if p is None:
        p = fm.FontProperties(fname=_FONT_FILES["ea" if ea else "lt"]["b" if bold else "n"])
        p.set_size(sz)
        _FP_CACHE[key] = p
    return p
def mixed(target, x, y, s, fontsize=9, color=TXT, bold=False,
          ha="center", va="bottom", zorder=10):
    spans = []
    for ch in str(s):
        cjk = bool(_CJK_RE.match(ch))
        if spans and spans[-1][1] == cjk:
            spans[-1][0] += ch
        else:
            spans.append([ch, cjk])
    if len(spans) == 1:
        t, cjk = spans[0]
        return [target.text(x, y, t, fontproperties=fp(fontsize, bold, cjk),
                            color=color, ha=ha, va=va, zorder=zorder)]
    renderer = target.figure.canvas.get_renderer()
    ws = []
    for t, cjk in spans:
        a = target.text(0, 0, t, fontproperties=fp(fontsize, bold, cjk), alpha=0)
        ws.append(a.get_window_extent(renderer).width)
        a.remove()
    tr = target.transData
    ax_, ay_ = tr.transform((x, y))
    x0 = ax_ - sum(ws) / 2.0 if ha == "center" else ax_
    inv = tr.inverted()
    out = []
    for (t, cjk), w in zip(spans, ws):
        fx, fy = inv.transform((x0 + w / 2.0, ay_))
        out.append(target.text(fx, fy, t, fontproperties=fp(fontsize, bold, cjk),
                               color=color, ha="center", va=va, zorder=zorder))
        x0 += w
    return out
def vgrad_bar(ax, x, h, w, c_top, c_bottom, zorder=3, n=240):
    c1 = np.array(mcolors.to_rgb(c_bottom))
    c2 = np.array(mcolors.to_rgb(c_top))
    t = np.linspace(0, 1, n).reshape(-1, 1, 1)
    img = c1.reshape(1, 1, 3) * (1 - t) + c2.reshape(1, 1, 3) * t
    return ax.imshow(img, extent=[x - w / 2, x + w / 2, 0, h], origin="lower",
                     aspect="auto", zorder=zorder, interpolation="bilinear")
def fmtk(x, _=None):
    return "%d" % int(round(x))
TITLE_RUNS = [                       
    {   
        "sz": 20.0, "bold": True, "color": "#FFFFFF", "lnspc": 24.0,
        "runs": ["3", "月各区域销量分布"],
    },
    {   
        "sz": 14.0, "bold": False, "color": "#FFFFFF", "lnspc": 24.0,
        "runs": ["东北销量最多占比总销量的", "22%", "，华南销量最低"],
    },
]
NOTE_PARA = {                        
    "sz": 8.0, "bold": False, "color": NOTE,
    "runs": ["*", "注：数据来源于公司销售系统，统计日期", "截至", "2022.03.31"],
    "run_dx": [0.0, 0.0, 1.5, 2.5],
}
BOX_TITLE = [0.06036, 0.06209, 0.95742, 0.26225]
BOX_NOTE = [0.04198, 0.91625, 0.84345, 0.98121]
INSET_PT = (0.05, 0.05, 0.1, 0.1)  
TB_TWEAK = {0: [-12.8, -0.5], 1: [0.75]}
def draw_textboxes(fig, W, H):
    renderer = fig.canvas.get_renderer()
    SC = DPI / float(fig.dpi)         
    def measure(text, sz, bold):
        a = fig.text(0, 0, text, fontproperties=fp(sz, bold, ea=True),
                     alpha=0, zorder=-1)
        w = a.get_window_extent(renderer).width * SC
        a.remove()
        return w
    boxes = [
        (BOX_TITLE, TITLE_RUNS),
        (BOX_NOTE, [NOTE_PARA]),
    ]
    for bi, (box, paras) in enumerate(boxes):
        x0, y0, x1, y1 = box
        lI = INSET_PT[2] * DPI                    
        tI = INSET_PT[0] * DPI                     
        xL = x0 * W + lI
        y = y0 * H + tI                            
        for pi, p in enumerate(paras):
            sz, bold = p["sz"], p["bold"]
            em = sz * EM_PT
            lh = (p.get("lnspc") or 1.71 * sz) * EM_PT
            off = TB_TWEAK.get(bi, [0.0] * 9)[pi] if pi < len(TB_TWEAK.get(bi, [])) else 0.0
            ytop = y + 0.43 * em + off       
            x = xL
            rdx = p.get("run_dx") or []
            for ri, text in enumerate(p["runs"]):
                xr = x + (rdx[ri] if ri < len(rdx) else 0.0)
                fig.text(xr / W, 1.0 - ytop / H, text,
                         fontproperties=fp(sz, bold, ea=True),   
                         color=p["color"], ha="left", va="top", zorder=30)
                x += measure(text, sz, bold)
            y += lh
def draw():
    W, H = SIZE_PX
    fig = plt.figure(figsize=(W / DPI, H / DPI), dpi=DPI)
    fig.patch.set_facecolor(BG)
    px, py, pw, ph = PLOT_PX
    ax = fig.add_axes([px, 1.0 - (py + ph), pw, ph])
    ax.set_facecolor(BG)
    ymax = 4000
    ax.set_ylim(0, ymax)
    ax.set_xlim(-0.6, len(VALS) - 0.4)
    for i, v in enumerate(VALS):
        vgrad_bar(ax, i, v, 0.32, BLUE, CYAN)    
        mixed(ax, i, v + ymax * 0.03, fmtk(v), fontsize=9, color=TXT,
              bold=True, ha="center", va="bottom")
    for side in ("top", "right", "left"):
        ax.spines[side].set_visible(False)
    ax.spines["bottom"].set_visible(False)         
    ax.set_yticks([0, 1000, 2000, 3000, 4000])
    ax.yaxis.set_major_formatter(FuncFormatter(fmtk))
    ax.tick_params(axis="both", length=0, labelsize=9, colors=TXT)
    ax.set_axisbelow(True)
    ax.set_xticks(range(len(CATS)))
    ax.set_xticklabels(CATS, fontsize=10, color=TXT)
    for lb in list(ax.get_xticklabels()) + list(ax.get_yticklabels()):
        lb.set_fontproperties(fp(lb.get_fontsize(), False, ea=bool(_CJK_RE.search(lb.get_text()))))
    ax.grid(axis="y", color=GRID, linewidth=0.75, linestyle=(0, (7.2, 1.8)))
    x0, x1 = ax.get_xlim()
    ax.plot([x0, x1], [0, 0], color=GRID, lw=0.75, ls=(0, (7.2, 1.8)),
            solid_capstyle="butt", clip_on=False, zorder=2.5)
    draw_textboxes(fig, W, H)
    return fig
