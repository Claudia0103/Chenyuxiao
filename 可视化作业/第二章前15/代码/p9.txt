import os
import numpy as np
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
from matplotlib import font_manager as fm
from matplotlib.patches import FancyArrowPatch
try:
    HERE = os.path.dirname(os.path.abspath(__file__))
except NameError:
    HERE = os.path.abspath(os.getcwd())
OUT = os.path.join(HERE, "png")
os.makedirs(OUT, exist_ok=True)
DPI = 120.0
EM_PT = DPI / 72.0
SIZE_PX = [700, 560]
BG = "#1A1E43"
BLUE = "#0070C0"
LBLUE = "#8FAADC"
TXT = "#FFFFFF"
LBL = "#F2F2F2"
NOTE = "#BFBFBF"
LINE = "#8FAADC"
ARROW = "#FFFFFF"
CATS = ["口红", "面膜", "隔离", "防晒", "精华"]
Y21 = [3568.0, 4135.0, 4436.0, 4106.0, 4936.0]
Y22 = [2569.0, 3241.0, 2965.0, 3209.0, 3541.0]
DIFF = [c - d for c, d in zip(Y21, Y22)]
VMAX = 5600.0
PLOT_L = 65.0
PLOT_R = 660.0
PLOT_B = 95.0
PLOT_T = 420.0
PLOT_H = PLOT_T - PLOT_B
GROUP_CX = [115.0, 250.0, 385.0, 520.0, 630.0]
BAR_W_PX = 38.0
BAR_GAP_PX = 5.0
TICK_LEN_PX = 28.0
def v_to_px(v):
    return PLOT_B + (v / VMAX) * PLOT_H
_FONT_FILES = {
    "ea": {"n": "C:/Windows/Fonts/msyh.ttc", "b": "C:/Windows/Fonts/msyhbd.ttc"},
    "cal": {"n": "C:/Windows/Fonts/calibri.ttf", "b": "C:/Windows/Fonts/calibrib.ttf"},
    "num": {"n": "C:/Windows/Fonts/arial.ttf", "b": "C:/Windows/Fonts/arialbd.ttf"},
}
for _d in _FONT_FILES.values():
    for _p in _d.values():
        try:
            fm.fontManager.addfont(_p)
        except Exception:
            pass
_FP_CACHE = {}
def fp(sz, bold=False, fam="ea"):
    key = (round(float(sz), 2), bool(bold), fam)
    p = _FP_CACHE.get(key)
    if p is None:
        p = fm.FontProperties(fname=_FONT_FILES[fam]["b" if bold else "n"])
        p.set_size(sz)
        _FP_CACHE[key] = p
    return p
def fmtk(x):
    return "%d" % int(round(x))
BOX_TITLE = [0.085, 0.04, 0.86, 0.20]
BOX_NOTE = [0.085, 0.90, 0.86, 0.98]
BOX_LEGEND = [0.085, 0.235]

TITLE_LINES = [
    {
        "sz": 20.0, "bold": True, "color": TXT, "fam": "ea",
        "lh_k": 1.70, "tweak": 0.25,
        "runs": ["2022", "年商品对比去年销售情况"],
        "run_dx": [0.0, 0.0],
    },
    {
        "sz": 11.5, "bold": False, "color": TXT, "fam": "ea",
        "lh_k": 1.75, "tweak": 1.0, "dx": 0.0,
        "runs": ["商品整体比去年销量有所下降，其中隔离下降最多，下降", "33%"],
        "run_dx": [0.0, 0.0],
    },
]
NOTE_LINES = [
    {
        "sz": 8.5, "bold": False, "color": NOTE, "fam": "ea",
        "lh_k": 1.75, "tweak": 0.0,
        "runs": ["*注：数据来源于公司销售系统，统计日期截至", "2022.01.01"],
        "run_dx": [0.0, 0.0],
    },
]
def draw_textboxes(fig, W, H):
    renderer = fig.canvas.get_renderer()
    SC = DPI / float(fig.dpi)
    def measure(text, sz, bold, fam):
        a = fig.text(0, 0, text, fontproperties=fp(sz, bold, fam), alpha=0, zorder=-1)
        w = a.get_window_extent(renderer).width * SC
        a.remove()
        return w
    for box, lines in ((BOX_TITLE, TITLE_LINES), (BOX_NOTE, NOTE_LINES)):
        x = box[0] * W
        y = box[1] * H
        for ln in lines:
            sz, bold, fam = ln["sz"], ln["bold"], ln["fam"]
            em = sz * EM_PT
            xr0 = x + ln.get("dx", 0.0)
            ytop = y + 0.43 * em + ln["tweak"]
            for ri, text in enumerate(ln["runs"]):
                rdx = ln.get("run_dx") or []
                fig.text(
                    (xr0 + (rdx[ri] if ri < len(rdx) else 0.0)) / W,
                    1.0 - ytop / H,
                    text,
                    fontproperties=fp(sz, bold, fam),
                    color=ln["color"],
                    ha="left", va="top", zorder=30,
                )
                xr0 += measure(text, sz, bold, fam)
            y += ln["lh_k"] * sz * EM_PT
def draw_legend(fig, W, H):
    x0 = BOX_LEGEND[0] * W
    y0 = BOX_LEGEND[1] * H
    fig.patches.append(
        plt.Rectangle(
            (x0 / W, 1.0 - (y0 + 9.0) / H),
            9.0 / W, 9.0 / H,
            transform=fig.transFigure,
            facecolor=BLUE, edgecolor="none", zorder=30,
        )
    )
    fig.text(
        (x0 + 14.0) / W, 1.0 - (y0 + 9.0) / H,
        "2021销量",
        fontproperties=fp(10.0, False, "ea"),
        color=TXT, ha="left", va="bottom", zorder=30,
    )
    x1 = x0 + 90.0
    fig.patches.append(
        plt.Rectangle(
            (x1 / W, 1.0 - (y0 + 9.0) / H),
            9.0 / W, 9.0 / H,
            transform=fig.transFigure,
            facecolor=LBLUE, edgecolor="none", zorder=30,
        )
    )
    fig.text(
        (x1 + 14.0) / W, 1.0 - (y0 + 9.0) / H,
        "2022销量",
        fontproperties=fp(10.0, False, "ea"),
        color=TXT, ha="left", va="bottom", zorder=30,
    )
def draw():
    W, H = SIZE_PX
    fig = plt.figure(figsize=(W / DPI, H / DPI), dpi=DPI)
    fig.patch.set_facecolor(BG)
    ax = fig.add_axes([0, 0, 1, 1])
    ax.set_xlim(0, W)
    ax.set_ylim(0, H)
    ax.set_facecolor(BG)
    for side in ("top", "right", "bottom", "left"):
        ax.spines[side].set_visible(False)
    ax.set_xticks([])
    ax.set_yticks([])
    ax.plot([PLOT_L, PLOT_R], [PLOT_B, PLOT_B], color=LINE, linewidth=0.8, zorder=2)
    for i in range(len(CATS)):
        cx = GROUP_CX[i]
        x_blue = cx - BAR_GAP_PX / 2.0 - BAR_W_PX
        x_lblue = cx + BAR_GAP_PX / 2.0
        h_blue = v_to_px(Y21[i]) - PLOT_B
        h_lblue = v_to_px(Y22[i]) - PLOT_B
        ax.add_patch(plt.Rectangle(
            (x_blue, PLOT_B), BAR_W_PX, h_blue,
            facecolor=BLUE, edgecolor="none", zorder=3,
        ))
        ax.add_patch(plt.Rectangle(
            (x_lblue, PLOT_B), BAR_W_PX, h_lblue,
            facecolor=LBLUE, edgecolor="none", zorder=3,
        ))
        ax.text(
            x_blue + BAR_W_PX / 2.0, PLOT_B + h_blue + 6.0, fmtk(Y21[i]),
            fontproperties=fp(9.5, False, "num"),
            color=TXT, ha="center", va="bottom", zorder=7,
        )
        ax.text(
            x_lblue + BAR_W_PX / 2.0, PLOT_B + h_lblue - 12.0, fmtk(Y22[i]),
            fontproperties=fp(9.5, False, "num"),
            color=TXT, ha="center", va="top", zorder=7,
        )
        x_tick0 = x_blue + BAR_W_PX
        x_tick1 = x_tick0 + TICK_LEN_PX
        y_tick = PLOT_B + h_blue
        ax.plot(
            [x_tick0, x_tick1], [y_tick, y_tick],
            color=BLUE, linewidth=2.0, zorder=6,
            solid_capstyle="butt",
        )
        y_start = PLOT_B + h_lblue
        y_end = y_tick
        x_arrow = x_tick1
        arrow = FancyArrowPatch(
            (x_arrow, y_start), (x_arrow, y_end),
            arrowstyle="-|>", mutation_scale=11,
            color=ARROW, linewidth=1.0, zorder=6,
        )
        ax.add_patch(arrow)
        ax.text(
            x_arrow + 6.0, (y_start + y_end) / 2.0, fmtk(DIFF[i]),
            fontproperties=fp(9.5, False, "num"),
            color=TXT, ha="left", va="center", zorder=7,
        )

        ax.text(
            cx, PLOT_B - 16.0, CATS[i],
            fontproperties=fp(10.5, False, "ea"),
            color=LBL, ha="center", va="top", zorder=7,
        )
    draw_legend(fig, W, H)
    draw_textboxes(fig, W, H)
    return fig
if __name__ == "__main__":
    f = draw()
    out = os.path.join(OUT, "chart09_对比柱形图.png")
    f.savefig(out, dpi=DPI, facecolor=BG)
    try:
        from IPython import get_ipython
        if get_ipython() is not None:
            from IPython.display import display
            display(f)
    except Exception:
        pass
    plt.close(f)
    print("已输出：", out)
