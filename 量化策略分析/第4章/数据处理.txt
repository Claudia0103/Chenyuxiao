# ---------------------------------------------------------------- 3 查询
def demo_query():
    s = pd.DataFrame({"group": [1, 1, 1, 2, 1, 2],
                      "name": ["Bob", "Lindy", "Mark", "Miki", "Sully", "Rose"],
                      "score": [98, 78, 87, 77, 65, 67]})
    show("3.1 布尔条件 score>70", s.score > 70)
    show("3.2 布尔索引", s[s.score > 70])
    show("3.3 且 &", s[(s.score > 70) & (s.group == 1)])
    show("3.4 非 ~", s[~(s.group == 1)])
    show("3.5 或 |", s[(s.group == 2) | (s.group == 1)])
    show("3.6 query('score > 90')", s.query("score > 90"))
    show("3.7 query 带逻辑", s.query("(group == 2) | (group == 1)"))
    show("3.8 between(70,80, inclusive='both')",
         s[s["score"].between(70, 80, inclusive="both")])
    show("3.9 isin", s[s["name"].isin(["Bob", "Lindy"])])
    show("3.10 str.contains 正则 [M]+", s[s["name"].str.contains("[M]+")])


# ---------------------------------------------------------------- 4 横向连接
def demo_merge():
    d1 = pd.DataFrame({"col1": ["a", "b", "c"], "id": [1, 2, 3]})
    d2 = pd.DataFrame({"col2": ["d", "e"], "id": [4, 3]})
    show("4.1 左表 / 右表", "d1=\n" + str(d1) + "\nd2=\n" + str(d2))
    for how in ("inner", "left", "right", "outer"):
        show(f"4.2 merge how='{how}'", d1.merge(d2, how=how, on="id"))

    a = pd.DataFrame({"id1": [1, 2, 3], "col1": ["a", "b", "c"]}, index=[1, 2, 3])
    b = pd.DataFrame({"id2": [1, 2, 3], "col2": ["aa", "bb", "cc"]}, index=[1, 3, 2])
    show("4.3 concat(axis=1) 按索引对齐", pd.concat([a, b], axis=1))
    show("4.4 join 按索引对齐", a.join(b))


# ---------------------------------------------------------------- 5 纵向合并
def demo_concat():
    d1 = pd.DataFrame({"id": [1, 1, 1, 2, 3, 4, 6],
                       "col": ["a", "a", "b", "c", "v", "e", "q"]})
    d2 = pd.DataFrame({"id": [1, 2, 3, 3, 5], "col": ["x", "y", "z", "v", "w"]})
    show("5.1 纵向拼接", pd.concat([d1, d2], ignore_index=True))
    show("5.2 拼接后 drop_duplicates",
         pd.concat([d1, d2], ignore_index=True).drop_duplicates())
    d3 = d1.rename(columns={"col": "new_col"})
    show("5.3 列名不同时拼接（产生新列）",
         pd.concat([d1, d3], ignore_index=True).drop_duplicates())

# ---------------------------------------------------------------- 7 分组汇总
def demo_groupby():
    s = pd.DataFrame({"name": ["Bob", "Lindy", "Mark", "Miki", "Sully", "Rose"],
                      "class": [1, 1, 1, 2, 1, 2], "grade": [1, 1, 1, 2, 2, 2],
                      "chinese": [88, 78, 86, 56, 77, 54],
                      "math": [98, 78, 87, 77, 77, np.nan]})
    show("7.1 单组最大值", s.groupby("grade")[["math"]].max())
    show("7.2 双键均值", s.groupby(["grade", "class"])[["math"]].mean())
    show("7.3 多列均值", s.groupby("grade")[["math", "chinese"]].mean())
    show("7.4 agg 多个统计量", s.groupby("class")["math"].agg(["mean", "min", "max"]))

    df = s.groupby(["grade", "class"])[["math", "chinese"]].agg(["min", "max"])
    show("7.5 多层列索引结果", df)
    show("7.6 取 ('math','min') 一列", df.loc[:, ("math", "min")])
    show("7.7 取某个单元格", df.loc[(2, 2), ("math", "min")])

# ---------------------------------------------------------------- 9 赋值
def demo_assign():
    s = pd.DataFrame({"name": ["Bob", "Lindy", "Mark", "Miki", "Sully", "Rose"],
                      "score": [99, 78, 999, 77, 77, np.nan],
                      "group": [1, 1, 1, 2, 1, 2]})
    show("9.1 replace 单列", s.score.replace(999, np.nan))
    show("9.2 replace 字典按列替换", s.replace({"score": {999: np.nan}, "name": {"Bob": np.nan}}))

    def transform(row):
        if row["group"] == 1:
            return "class1"
        return "class2"

    show("9.3 apply(axis=1)", s.apply(transform, axis=1))
    show("9.4 assign 接 apply", s.assign(class_n=s.apply(transform, axis=1)))
    t = s.copy()
    t.loc[t.group == 1, "class_n"] = "class1"
    t.loc[t.group == 2, "class_n"] = "class2"
    show("9.5 loc 条件赋值（推荐写法）", t)

# ---------------------------------------------------------------- 12 噪声与分箱
def cap(x, quantile=(0.01, 0.99)):
    """盖帽法：把小于 1% 分位数、大于 99% 分位数的值拉回分位数上。"""
    q01, q99 = x.quantile(list(quantile)).tolist()
    y = x.copy()
    if q01 > x.min():
        y.loc[y < q01] = q01
    if q99 < x.max():
        y.loc[y > q99] = q99
    return y


def demo_noise():
    np.random.seed(0)
    raw = pd.DataFrame({"normal": np.random.randn(1000)})
    capped = raw.apply(cap, quantile=[0.01, 0.99])
    show("12.1 盖帽前后极值与标准差", pd.DataFrame({
        "原始": [raw.normal.min(), raw.normal.max(), raw.normal.std()],
        "盖帽后": [capped.normal.min(), capped.normal.max(), capped.normal.std()]},
        index=["min", "max", "std"]).round(4))

    fig, ax = plt.subplots(1, 2, figsize=(10, 3.8))
    ax[0].hist(raw.normal, bins=50, color="#5aa9ff")
    ax[0].set_title("盖帽前")
    ax[1].hist(capped.normal, bins=50, color="#3ddc97")
    ax[1].set_title("盖帽后（1%~99% 分位）")
    for a in ax:
        a.set_xlabel("取值"); a.set_ylabel("频数")
    fig.tight_layout()
    p = os.path.join(CHARTS, "4.1_盖帽法前后对比.png")
    fig.savefig(p, dpi=120); plt.close(fig)
    print(f"12.2 直方图已保存：{p}")

    np.random.seed(0)
    s = pd.DataFrame({"normal": np.random.randn(10)})
    show("12.3 等宽分 5 箱", pd.cut(s.normal, 5))
    show("12.4 自定义标签", pd.cut(s.normal, bins=5, labels=[1, 2, 3, 4, 5]))
    show("12.5 两箱 bad/good", pd.cut(s.normal, bins=2, labels=["bad", "good"]))
    show("12.6 分位数", s.normal.quantile([0, 0.5, 1]).round(4))
    show("12.7 按分位数分箱",
         pd.cut(s.normal, bins=s.normal.quantile([0, 0.5, 1]),
                labels=["bad", "good"], include_lowest=True))
